def test_existing_stop_reissues_action_after_unknown_runtime_error():
    calls=[];tick=[0]
    def invoke(service,action):
        calls.append((service,action))
        if len(calls)==1:raise RuntimeError('SYNTHETIC UNKNOWN MANAGED STOP')
        return {'status':'stopped'}
    fake=SimpleNamespace(clock=lambda:tick[0],sleep=lambda n:tick.__setitem__(0,tick[0]+n),invoke=invoke)
    assert AdaptiveRoll.stop(fake,'engine')=={'status':'stopped'}
    assert calls==[('engine','stop'),('engine','stop')]

def test_existing_activation_rolls_back_and_restarts_watchdog_after_unknown_start():
    fake,actions=fixture()
    with pytest.raises(TimeoutError):CollectiveRoll.apply(fake)
    failed=actions.index(('start_engine',fake.candidate_engine))
    later=actions[failed+1:]
    assert ('start_engine',fake.engine) in later
    assert ('start',fake.worker) in later and ('start',fake.watchdog) in later
    assert fake.report['phase']=='rolled_back'

def test_existing_restore_performs_followon_exclusion_after_unknown_stop():
    fake,actions=fixture();original=fake.stop;once=[False]
    def stop(service):
        original(service)
        if service==fake.candidate_engine and not once[0]:once[0]=True;raise TimeoutError('SYNTHETIC UNKNOWN ENGINE STOP')
    fake.stop=stop
    with pytest.raises(TimeoutError):RestoreCollective.apply(fake)
    failed=actions.index(('stop',fake.candidate_engine));later=actions[failed+1:]
    assert ('manifest',fake.original_engine_only) in later
    assert ('stop',fake.worker) in later and ('start',fake.watchdog) in later
    assert fake.report['worker_excluded_and_owned_worker_stopped'] is True
