def private(path,raw):
    path.parent.mkdir(parents=True,exist_ok=True,mode=0o700);path.write_bytes(raw);path.chmod(0o600)
def jsonfile(path,value):
    private(path,M['encoded'](value));return {'path':str(path),'sha256':M['sha'](path.read_bytes())}

class Clock:
    def __init__(self):self.now=0
    def __call__(self):return self.now
    def sleep(self,seconds):self.now+=seconds

class Controller:
    def __init__(self,root,services,watch,clock):
        self.root,self.clock,self.events=root,clock,[];self.next=100
        self.records={};self.environ={};self.argv={};self.quiet=True;self.worker_ready_at=0
        self.fail_stage=None;self.probe=None
        for n,s in enumerate([*services,watch]):
            key=self.key(s);self.records[key]={'pid':n+10,'start_ticks':str(n+50),'cwd':str(root),'command':['--key','NEVER-PERSIST-process-command']}
            self.environ[key]={**M['values'](s['env_file'] and Path(s['env_file']).read_bytes()),'AMBIENT':'NEVER-PERSIST-ambient'}
            self.argv[key]=('python-'+key).encode()
    @staticmethod
    def key(s):return s['role']+'-'+s['name']
    def record(self,s):return self.records.get(self.key(s))
    def responding(self,s,ready=False):
        return self.record(s) is not None and (s['role']!='worker' or self.clock()>=self.worker_ready_at)
    def engine_quiescence(self,s):
        self.events.append(('engine','GET_metrics'))
        return {'quiescent':self.quiet,'error':None if self.quiet else 'metrics_unavailable'}
    def read(self,controller,s):
        key=self.key(s);r=self.record(s);M['require'](r is not None,'owned_service_missing')
        env=self.environ[key];argv=self.argv[key]
        return r,dict(env),argv,{k:r[k] for k in ('pid','start_ticks','cwd')}|{
            'process_environ_sha256':M['sha'](M['encoded'](env)),'process_argv_sha256':M['sha'](argv)}
    def intent(self,stage):
        saved=json.loads(self.probe.report_path.read_bytes())
        assert saved['pending_action']['stage']==stage and saved['pending_action']['nonce']
    def graceful_stop(self,s,expected,grace,check):
        assert self.record(s)==expected;stage='watchdog_stop' if s['role']=='supervisor' else 'idle_worker_stop'
        self.intent(stage);check();self.events.append((self.key(s),'stop'));self.records.pop(self.key(s))
        if self.fail_stage==stage:raise OSError('NEVER-PERSIST-stop-error')
    def start(self,root,s,environment):
        stage='worker_start' if s['role']=='worker' else 'watchdog_start';self.intent(stage)
        assert self.record(s) is None
        self.next+=1;key=self.key(s);self.events.append((key,'start'))
        self.records[key]={'pid':self.next,'start_ticks':str(self.next*10),'cwd':str(root),'command':['--key','NEVER-PERSIST-new-command']}
        self.environ[key]={**environment,**M['values'](Path(s['env_file']).read_bytes())}
        if self.fail_stage==stage:raise TimeoutError('NEVER-PERSIST-start-ack')
        return {'status':'started','pid':self.next}

    def test_exact_stop_callback_allows_draining_503_and_exit_before_pidfile_unlink(self):
        with tempfile.TemporaryDirectory() as d:
            job,c,_=self.setup(d);read=c.record;inside=[None];observed=[]
            def record(service):
                if service is inside[0]:raise AssertionError('target sampled outside pidfd owner')
                return read(service)
            c.record=record
            def graceful_stop(service,expected,grace,check):
                self.assertEqual(read(service),expected)
                c.intent('watchdog_stop' if service is job.watchdog else 'idle_worker_stop')
                # Model the callback after exact role lock + pidfd acquisition.
                # A draining worker returns ready503; after kernel exit /proc
                # vanishes while the unchanged PID file is still present.
                if service is job.worker:c.worker_ready_at=1000
                inside[0]=service
                try:check();observed.append((c.key(service),'draining'))
                finally:inside[0]=None
                c.records.pop(c.key(service));c.events.append((c.key(service),'stop'))
                inside[0]=service
                try:check();observed.append((c.key(service),'exited_pidfile_pending'))
                finally:inside[0]=None
                if service is job.worker:c.worker_ready_at=0
                # The real helper checks the original PID file before unlink.
            c.graceful_stop=graceful_stop
            self.assertEqual(job.apply()['phase'],'complete')
            self.assertEqual(len(observed),4)
