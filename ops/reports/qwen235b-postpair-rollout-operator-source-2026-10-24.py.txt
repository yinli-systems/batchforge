def require(value,code):

def sha(raw):return hashlib.sha256(raw).hexdigest()

def encoded(value):return (json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False)+'\n').encode()
def load(raw):
    def unique(pairs):
        value={}
        for k,v in pairs:require(k not in value,'duplicate_json_key');value[k]=v
        return value
    def bad(_):raise RuntimeError('nonfinite_json')
    return json.loads(raw.decode('utf-8'),object_pairs_hook=unique,parse_constant=bad)
def fsync_directory(path):

def snapshot_state(path):

        and info.st_nlink==1 and stat.S_IMODE(info.st_mode)==0o600 and 0<info.st_size<=512*1024**2,'unsafe_snapshot_state')
    return {k:getattr(info,k) for k in ('st_dev','st_ino','st_size','st_mtime_ns','st_ctime_ns')}

def bundle(path,expected):
    raw=private_file(path,4*1024**2);require(sha(raw)==expected,'bundle_sha_differs');sources={}
    with tarfile.open(fileobj=io.BytesIO(raw),mode='r:*') as archive:
        for m in archive:
            require(m.name in TARGETS and m.name not in sources and m.isreg() and 0<=m.size<=512*1024,'exact_eleven_source_bundle_required')
            source=archive.extractfile(m).read(512*1024+1);require(len(source)==m.size and sha(source)==TARGETS[m.name],'reviewed_source_sha_differs')
            if m.name.endswith('.py'):compile(source,m.name,'exec',dont_inherit=True)
            sources[m.name]=source
    require(set(sources)==set(TARGETS),'incomplete_source_bundle');return sources

def shutdown_bytes(raw):
    lines=[];seen=0
    for line in raw.decode().splitlines(keepends=True):
        if re.match(r'^\s*(?:export\s+)?WORKER_SHUTDOWN_SECONDS\s*=',line):
            seen+=1;require(seen==1,'duplicate_shutdown_assignment');lines.append('WORKER_SHUTDOWN_SECONDS=600\n')
        else:lines.append(line)
    if not seen:
        if lines and not lines[-1].endswith(('\n','\r')):lines.append('\n')
        lines.append('WORKER_SHUTDOWN_SECONDS=600\n')
    patched=''.join(lines).encode();old=values(raw);new=values(patched)
    require(new.get('WORKER_SHUTDOWN_SECONDS')=='600' and {k:v for k,v in new.items() if k!='WORKER_SHUTDOWN_SECONDS'}==
        {k:v for k,v in old.items() if k!='WORKER_SHUTDOWN_SECONDS'},'non_shutdown_environment_change')
    return patched

def scoped_caches(root,remove=False):
    entries=[]
    for name in sorted(k for k in TARGETS if k.endswith('.py')):
        source=root/name;folder=source.parent/'__pycache__'
        if not folder.exists() and not folder.is_symlink():continue
        info=folder.lstat();require(stat.S_ISDIR(info.st_mode) and info.st_uid==os.geteuid() and not info.st_mode&0o7022,'unsafe_scoped_cache_directory')
        pattern=re.compile(re.escape(source.stem)+r'\.[A-Za-z0-9_-]+(?:\.opt-[0-9]+)?\.pyc\Z')
        for path in folder.iterdir():
            if not pattern.fullmatch(path.name):continue
            info=path.lstat();require(stat.S_ISREG(info.st_mode) and info.st_uid==os.geteuid() and not info.st_mode&0o7022,'unsafe_scoped_cache')
            entries.append((path,info.st_dev,info.st_ino))
    if remove:
        for path,device,inode in entries:
            info=path.lstat();require((info.st_dev,info.st_ino)==(device,inode),'scoped_cache_replaced');path.unlink()
        for folder in {p.parent for p,_,_ in entries}:fsync_directory(folder)
    return [str(p.relative_to(root)) for p,_,_ in entries]

def process_data(controller,service):
    record=controller.record(service);require(record is not None,'owned_service_missing')
    process=Path('/proc')/str(record['pid']);raw=(process/'environ').read_bytes();argv=(process/'cmdline').read_bytes()

    return record,environment,argv,{'pid':record['pid'],'start_ticks':record['start_ticks'],'cwd':record['cwd'],
        'process_environ_sha256':sha(raw),'process_argv_sha256':sha(argv)}

def invoke_start(root,service,environment):
    """Reuse manage.py protocol, preserving the original effective environment."""
    argv=[service['python'],str(root/'ops/manage.py'),'start',service['role'],'--name',service['name'],
        '--env-file',service['env_file'],'--python',service['python']]
    for key in ('port','gpu'):
        if service.get(key) is not None:argv.extend(('--'+key,str(service[key])))
    result=subprocess.run(argv,cwd=root,env=environment,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,
        stderr=subprocess.DEVNULL,timeout=30,check=False)
    require(result.returncode==0 and len(result.stdout)<=4096,'managed_start_unconfirmed')
    value=load(result.stdout);require(isinstance(value,dict),'manager_protocol_invalid');return value

    def private_guard(self):
        for name in TARGETS:
            require(private_file(self.run/'staged'/name,512*1024)==self.sources[name],'staged_source_changed')
            if self.original[name] is not None:require(private_file(self.run/'original'/name,512*1024)==self.original[name],'original_source_snapshot_changed')
        require(private_file(self.run/'worker.env.original')==self.original_worker_env
            and private_file(self.run/'worker.env.staged')==self.patched_worker_env,'worker_private_snapshot_changed')
    def guard(self,*,allow_unready_worker=False,stopping_target=None):
        require(not self.unknown and not self.stop.is_set(),'unknown_or_interrupted_operation')
        if stopping_target is not None:
            # Only graceful_stop invokes this callback, after acquiring its
            # role lock and exact pidfd. That helper owns target identity/exit
            # checks until it unlinks the unchanged original PID record.
            require(stopping_target is self.worker or stopping_target is self.watchdog,'invalid_stopping_target')
            pending=self.report['pending_action'];expected=self.expected_ids[self.key(stopping_target)]
            stage='idle_worker_stop' if stopping_target is self.worker else 'watchdog_stop'
            require(pending is not None and pending['stage']==stage and expected is not None
                and pending['pid']==expected['pid'] and pending['start_ticks']==expected['start_ticks'],'stopping_target_not_exact_pending_action')
        require(private_file(self.manifest)==self.manifest_raw,'manifest_drift')
        require(sha(SELF.read_bytes())==self.report['binding']['controller_sha256'],'controller_source_drift')
        require(dependency_hashes(self.root)==self.dependencies,'dependency_source_drift')
        require(all(private_file(Path(p))==v for p,v in self.env.items()),'environment_drift')
        require(all(sha(private_file(Path(p)))==digest for p,digest in self.auxiliary_configuration.items()),'auxiliary_configuration_drift')
        require(all(source_state(self.root/p)[0]==state for p,state in self.expected_states.items()),'source_drift')
        require(all(self.controller.record(s)==self.expected_ids[self.key(s)] for s in self.all if s is not stopping_target),'foreign_process_identity')
        for service in self.all:
            key=self.key(service)
            if service is stopping_target or self.expected_ids[key] is None:continue
            record,environment,argv,_=self.process_reader(self.controller,service)
            require(record==self.expected_ids[key] and environment==self.expected_process_env[key]
                and argv==self.process_argv[key],'effective_process_binding_drift')
        for service in self.services:
            if service is not stopping_target and self.expected_ids[self.key(service)] is not None and not (allow_unready_worker and service==self.worker):
                require(self.controller.responding(service,ready=True),'service_not_ready')
        self.private_guard();self.drained()
        observed=datetime.fromisoformat(self.proofs['backup']['observed_at']);require(0<=(datetime.now(timezone.utc)-observed).total_seconds()<=3600,'backup_expired')
        require(snapshot_state(Path(self.proofs['backup']['snapshot_path']))==self.proofs['backup']['snapshot_state'],'snapshot_changed_after_verification')
        # Proof files are immutable. Rehash small metadata, not the large dump
        # for each poll. Full dump verification occurs at prepare AND apply.
        def check(value):
            if isinstance(value,dict) and set(value)=={'path','sha256'}:checked_file(value,self.root)
            elif isinstance(value,dict):
                for item in value.values():check(item)
        check(self.proof_spec)
    def _drained(self):
        import psycopg
        from backend.migrate import MIGRATIONS
        dsn=values(self.env[self.api['env_file']])['DATABASE_URL']
        with psycopg.connect(dsn,autocommit=True,connect_timeout=5,options='-c default_transaction_read_only=on -c statement_timeout=5000 -c lock_timeout=1000') as conn:
            require(conn.execute('SELECT current_user,current_database()').fetchone()==('bf_runtime','batchforge'),'runtime_database_role_required')
            require(conn.execute('SELECT version,checksum FROM batchforge_schema_migrations ORDER BY version').fetchall()==[(v,h) for v,_,h in MIGRATIONS],'schema11_registry_changed')
            require(conn.execute("SELECT count(*) FROM tasks WHERE status IN ('pending','running')").fetchone()==(0,),'global_drain_required')
    def _profile(self):
        import psycopg
        from ops.ha.adaptive_qwen235b import metric_values
        expected={'batchforge_worker_capacity':64,'batchforge_worker_ready_engines':1,
            'batchforge_worker_adaptive_kv_budget_tokens':self.budget,'batchforge_worker_adaptive_kv_paused':0,
            'batchforge_worker_adaptive_kv_waiting':0,'batchforge_worker_inflight':0,'batchforge_worker_reserved_context_tokens':0}
        with self.controller.http.open(f"http://127.0.0.1:{self.worker['port']}/metrics",timeout=3) as response:
            require(response.status==200,'worker_metrics_unavailable');raw=response.read(2*1024**2+1)
        require(len(raw)<=2*1024**2 and metric_values(raw.decode(),expected)==expected,'worker_exact_kv_gauges_differ')
        with psycopg.connect(values(self.env[self.api['env_file']])['DATABASE_URL'],autocommit=True,connect_timeout=5,
                options='-c default_transaction_read_only=on -c statement_timeout=5000') as conn:
            row=conn.execute("SELECT concurrency,scheduler_profile_id,runtime_fingerprint,coalesce((metadata->>'ready_engine_count')::integer,0),metadata->'kv_admission',metadata->'scheduling_profile' FROM workers WHERE id=%s AND model=%s AND last_seen_at>clock_timestamp()-interval '60 seconds'",(self.worker_values['WORKER_ID'],MODEL)).fetchone()
        require(row is not None and row[:4]==(64,self.profile_value['profile_id'],self.profile_value['runtime_fingerprint'],1)
            and row[4]=={'protocol':'vllm-0.10.2-qwen-chat','budget':self.budget,'block_size':16,'single_worker_single_engine':True,'paused':False}
            and row[5]==self.profile_value,'worker_registration_profile_kv_differs')
        return {'capacity':64,'kv_budget':self.budget,'registration_verified':True}
    def quiet(self,stage,timeout=600):
        began=self.clock();observations=[]
        while True:
            self.guard();observation=self.controller.engine_quiescence(self.engine)
            observations.append({'at':stamp(),**observation})
            if observation['quiescent']:break
            require(self.clock()-began<timeout,'private_engine_quiescence_timeout');self.sleep(.5)
        self.report.setdefault('engine_quiescence',[]).append({'stage':stage,'wait_seconds':self.clock()-began,'observations':observations});self.save()
    def action(self,stage,call,**metadata):
        self.guard();require(self.report['pending_action'] is None,'pending_action_no_replay')
        self.report.update(phase='applying',pending_action={'stage':stage,'nonce':secrets.token_hex(16),'at':stamp(),**metadata},manual_recovery_required=True)
        self.save()
        intent=dict(self.report['pending_action'])
        try:
            result=call()
            self.report['events'].append({'stage':stage,'confirmed_at':stamp(),**metadata})
            self.report['pending_action']=None;self.save();return result
        except BaseException:
            self.unknown=True
            self.report.update(phase='manual_recovery_required',manual_recovery_required=True,
                pending_action={**intent,'outcome':'unknown_or_confirmation_not_durable'})
            try:self.save()
            except BaseException:pass
            raise
    def stop_service(self,service,stage):
        expected=self.expected_ids[self.key(service)]
        def stop():
            self.controller.graceful_stop(service,expected,30,lambda:self.guard(stopping_target=service))
            require(self.controller.record(service) is None,'stop_not_confirmed')
            self.expected_ids[self.key(service)]=None
        self.action(stage,stop,pid=expected['pid'],start_ticks=expected['start_ticks'])
        self.report['watchdog_state' if service==self.watchdog else 'worker_state']='confirmed_stopped';self.save()
    def start_service(self,service,stage):
        require(self.expected_ids[self.key(service)] is None and self.controller.record(service) is None,'start_requires_confirmed_absence')
        def start():
            result=self.starter(self.root,service,self.process_env[self.key(service)])
            record,environment,argv,binding=self.process_reader(self.controller,service)
            wanted=dict(self.process_env[self.key(service)])
            if service==self.worker:wanted['WORKER_SHUTDOWN_SECONDS']='600'
            require(result.get('status')=='started' and result.get('pid')==record['pid']
                and environment==wanted and argv==self.process_argv[self.key(service)],'start_ack_identity_or_effective_environment_unknown')
            self.expected_ids[self.key(service)]=record
            self.expected_process_env[self.key(service)]=wanted
            self.report['worker_after' if service==self.worker else 'watchdog_after']=binding
        self.action(stage,start,environment_sha256=sha(self.env[service['env_file']]))
        self.report['worker_state' if service==self.worker else 'watchdog_state']='known_started';self.save()
    def apply(self):
        require(self.report['phase']=='prepared' and self.report['pending_action'] is None,'existing_operation_no_replay')
        self.guard();require(self.proof_checker(self.root,self.proof_spec,self.sources)==self.proofs,'proof_binding_changed')
        self.quiet('before_idle_worker_stop');self.profile_check()
        try:
            self.stop_service(self.watchdog,'watchdog_stop')
            self.stop_service(self.worker,'idle_worker_stop')
            self.quiet('after_idle_worker_stop')
            for name,raw in sorted(self.sources.items()):
                def write_source(name=name,raw=raw):
                    state=self.expected_states[name];mtime=max(time.time_ns()+2_000_000_000,state.get('mtime_ns',0)+2_000_000_000)
                    self.writer(self.root/name,raw,state.get('mode',0o644),mtime)
                    current,_=source_state(self.root/name);require(current['exists'] and current['sha256']==TARGETS[name],'source_write_not_confirmed')
                    self.expected_states[name]=current
                self.action('source_copy',write_source,path=name,old_sha256=self.states[name].get('sha256'),new_sha256=TARGETS[name])
            def patch():
                atomic_bytes(Path(self.worker['env_file']),self.patched_worker_env)
                require(private_file(Path(self.worker['env_file']))==self.patched_worker_env,'worker_env_patch_not_confirmed')
                self.env[self.worker['env_file']]=self.patched_worker_env
            self.action('worker_shutdown_env_patch',patch,old_sha256=sha(self.original_worker_env),new_sha256=sha(self.patched_worker_env))
            self.action('scoped_bytecode_evict',lambda:self.report.update(bytecode_evicted=scoped_caches(self.root,True)))
            self.quiet('before_new_worker_start')
            self.start_service(self.worker,'worker_start')
            deadline=self.clock()+120
            while True:
                self.guard(allow_unready_worker=True)
                try:
                    require(self.controller.responding(self.worker,ready=True),'new_worker_not_ready')
                    self.report['worker_registration']=self.profile_check();break
                except Exception:
                    require(self.clock()<deadline,'new_worker_registration_timeout');self.sleep(.5)
            self.quiet('before_watchdog_restore');self.guard()
            self.start_service(self.watchdog,'watchdog_start');self.guard();self.profile_check()
            self.report.update(phase='complete',finished_at=stamp(),manual_recovery_required=False,watchdog_state='known_restored',worker_state='ready64',
                engine_configuration_changed=False,manifest_bytes_unchanged=True,protected_processes_unchanged=True)
            self.save();return self.report
        except BaseException as error:
            self.report.update(phase='manual_recovery_required',manual_recovery_required=True,error_type=type(error).__name__,
                no_automatic_rollback_or_service_restart=True)
            try:self.save()
            except BaseException:pass
            raise

@contextmanager
def operator_lock(root):
    fd=os.open(root/'.runtime/postpair-source-rollout.lock',os.O_RDWR|os.O_CREAT|os.O_NOFOLLOW,0o600)
    try:
        info=os.fstat(fd);require(stat.S_ISREG(info.st_mode) and info.st_uid==os.geteuid()
            and info.st_nlink==1 and not info.st_mode&0o077,'unsafe_operator_lock')
        fcntl.flock(fd,fcntl.LOCK_EX|fcntl.LOCK_NB);yield
    finally:os.close(fd)

def main(argv=None):
    os.umask(0o077)
    p=argparse.ArgumentParser(description=__doc__)
    p.add_argument('--manifest',type=Path,required=True);p.add_argument('--archive',type=Path,required=True)
    p.add_argument('--sha256',required=True);p.add_argument('--evidence-spec',type=Path,required=True)
    p.add_argument('--evidence-sha256',required=True);p.add_argument('--operation',required=True)
    p.add_argument('--prepared-sha256');p.add_argument('--apply',action='store_true')
    args=p.parse_args(argv);job=None
    try:
        require(sys.platform=='linux' and hasattr(os,'pidfd_open') and hasattr(signal,'pidfd_send_signal'),'linux_pidfd_required')
        require(re.fullmatch('[A-Za-z0-9][A-Za-z0-9_-]{0,63}',args.operation) is not None,'invalid_operation')
        sources=bundle(args.archive,args.sha256)
        proof_spec,_=checked_file({'path':str(args.evidence_spec),'sha256':args.evidence_sha256},ROOT)
        manifest,services,_=load_manifest(args.manifest,root=ROOT)
        namespace={'__name__':'postpair_approved_worker_controller','__file__':str(ROOT/'ops/ha/single_pod_fault_probe.py')}
        # The live base may still be4098. Load ONLY the pinned54b candidate into
        # RAM; do not install new serving dependencies merely to pause watchdog.
        exec(compile(sources['ops/ha/single_pod_fault_probe.py'],namespace['__file__'],'exec'),namespace)
        controller=namespace['WorkerController'](root=ROOT)
        run=ROOT/'.runtime'/args.operation
        directory(ROOT/'.runtime')
        with operator_lock(ROOT):
            if not run.exists():
                require(not args.apply,'apply_requires_prepared_operation');directory(run,fresh=True)
            else:directory(run)
            if args.apply:
                prepared=private_file(run/'report.json',16*1024**2)
                require(sha(prepared)==args.prepared_sha256 and load(prepared).get('phase')=='prepared','exact_prepared_sha_required')
            stop=threading.Event()
            for signum in (signal.SIGINT,signal.SIGTERM):signal.signal(signum,lambda *_:stop.set())
            job=Deployment(root=ROOT,run=run,manifest=manifest,services=services,controller=controller,
                sources=sources,bundle_sha=args.sha256,proof_spec=proof_spec,stop=stop)
            job.guard();job.quiet('prepared_preflight');job.profile_check()
            if args.apply:job.apply()
            print(json.dumps({'phase':job.report['phase'],'report':str(job.report_path),
                'report_sha256':sha(private_file(job.report_path,16*1024**2)),
                'worker_environment_keys_changed':job.report['changed_worker_environment_keys'],
                'engine_api_gateway_restart_attempted':False,'applied':args.apply},sort_keys=True))
            return 0
    except BaseException as error:
        # No raw manager output, command, environment, DSN or provider message.
        print(json.dumps({'phase':'refused_or_manual_recovery_required',
            'reason_code':str(error) if type(error) is RuntimeError else type(error).__name__,
            'automatic_replay':False,'report':str(job.report_path) if job else None},sort_keys=True))
        return 1
