class PolicyTests(unittest.TestCase):
    def test_success_only_policy_key_and_api_watchdog_pids_change(self):
        with tempfile.TemporaryDirectory() as d:
            f=Fixture(d);old=deepcopy(f.records);original=dict(f.roll.original);result=f.roll.apply()
            self.assertEqual(result['phase'],'complete');self.assertTrue(result['watchdog_restored'])
            self.assertEqual(f.events,['stop:supervisor-primary','stop:api-a','start:api-a','stop:api-b','start:api-b','start:supervisor-primary'])
            for k in ('worker-qwen235b','engine-qwen235b','gateway-edge'):self.assertEqual(f.records[k],old[k])
            for s in f.services:
                raw=Path(s['env_file']).read_bytes()
                self.assertEqual(raw,original[s['env_file']].replace(b'SCHEDULER_POLICY=cost_aware',b'SCHEDULER_POLICY=fair') if s['role']=='api' else original[s['env_file']])
            self.assertNotIn('SYNTHETIC_NEVER_REPORT',json.dumps(result))
            with self.assertRaises(ValueError):f.roll.apply()
    def test_unknown_stop_and_start_never_follow_on_or_restore_watchdog(self):
        for stage in ('stop:supervisor-primary','stop:api-a','start:api-a','stop:api-b','start:api-b','start:supervisor-primary'):
            with self.subTest(stage=stage),tempfile.TemporaryDirectory() as d:
                f=Fixture(d);f.failure=stage
                with self.assertRaises((TimeoutError,OSError)):f.roll.apply()
                self.assertEqual(f.roll.report['pending_action']['stage'],stage);self.assertEqual(f.roll.report['phase'],'manual_recovery_required')
                self.assertEqual(f.events[-1],stage);self.assertFalse(f.roll.report['watchdog_restart_after_failure'])
                before=list(f.events)
                with self.assertRaises(ValueError):f.roll.apply()
                self.assertEqual(f.events,before)
    def test_unknown_confirmation_fsync_preserves_intent(self):
        with tempfile.TemporaryDirectory() as d:
            f=Fixture(d);save=f.roll.save
            def failed(value):
                if value['events'] and value['events'][-1]['stage']=='stop:api-a' and value['pending_action'] is None:raise OSError('SYNTHETIC_FSYNC')
                save(value)
            f.roll.save=failed
            with self.assertRaises(OSError):f.roll.apply()
            self.assertEqual(f.roll.report['pending_action']['stage'],'stop:api-a');self.assertNotIn('start:supervisor-primary',f.events)
    def test_intent_fsync_failure_prevents_any_stop(self):
        with tempfile.TemporaryDirectory() as d:
            f=Fixture(d)
            f.roll.save=lambda _:(_ for _ in ()).throw(OSError('SYNTHETIC_FSYNC'))
            with self.assertRaises(OSError):f.roll.apply()
            self.assertEqual(f.events,[])

    def test_native_api_stop_uses_exact_pidfd_lock_term_and_original_pidfile(self):
        with tempfile.TemporaryDirectory() as d:
            f=Fixture(d);s=f.services[0];expected=f.record(s);path=f.root/'.runtime/api-a.json';private(path,H['encoded'](expected));calls=[]
            class Poll:
                def register(self,*args):calls.append(('register',args))
                def poll(self,*args):return [1]
            with patch.object(os,'pidfd_open',create=True,return_value=987),patch.object(M['signal'],'pidfd_send_signal',create=True,side_effect=lambda fd,sig:calls.append(('signal',fd,sig))),patch.object(M['select'],'poll',return_value=Poll()),patch.object(os,'close',side_effect=lambda fd:None if fd==987 else close(fd)):
