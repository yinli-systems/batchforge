class Child:
 def __init__(self,argv,log,spawn=subprocess.Popen):
  self.log=log;write(log,{'phase':'reserved','at':stamp(),'pipe_policy':'SHA256 and byte counts only; no plaintext'})
  self.process=spawn(argv,cwd=ROOT,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,start_new_session=True,
   env={**os.environ,'PYTHONDONTWRITEBYTECODE':'1'})
  self.pipes={};self.threads=[]
  def consume(name,pipe):
   digest=hashlib.sha256();size=0
   try:
    for chunk in iter(lambda:pipe.read(8192),b''):digest.update(chunk);size+=len(chunk)
    self.pipes[name]={'sha256':digest.hexdigest(),'bytes':size}
   except Exception:self.pipes[name]={'capture_error':'pipe_read_failed'}
   finally:pipe.close()
  for name in ('stdout','stderr'):
   thread=threading.Thread(target=consume,args=(name,getattr(self.process,name)),daemon=True);thread.start();self.threads.append(thread)
 def poll(self):return self.process.poll()
 def identity(self):
  from ops.manage import process_identity
  result=process_identity(self.process.pid);require(result and result['cwd']==str(ROOT),'owned_child_start_identity_unknown');return result
 def finish(self):
  require(self.poll() is not None,'child_not_terminal')
  for thread in self.threads:thread.join(2)
  require(all(not t.is_alive() for t in self.threads) and all('sha256' in value for value in self.pipes.values()),'child_pipe_outcome_unknown')
  result={'phase':'terminal','returncode':self.poll(),'pipes':self.pipes};write(self.log,result,exclusive=False);return result

    directory(parent);directory(self.run,fresh=True);sync(parent)

 def wait(self,child,timeout,context,binding,stage):

  result=child.finish();self.journal.setdefault('child_outcomes',{})[stage]=result;self.save()
  require(result['returncode']==0,'child_failed_no_automatic_replay');return result
