"""Same-Pod post-pair schema11 snapshot and isolated PG17 restore; no data export."""
from pathlib import Path
from datetime import datetime, timezone
import argparse, hashlib, importlib.util, json, os, subprocess, sys

ROOT = Path('/root/batchforge-runtime/app')
OPERATIONS = ('schema11-pre-fault-20261024-v1', 'schema11-pre-history-20261024-v1')
EXPORT_OPERATOR_SHA = '36336209d22979e88be63bebe8318a6536c553095a9a0d99aba998adaffb9438'
PAIR_JOURNAL_LIMIT = 2 * 1024**2
sys.path.insert(0, str(ROOT)); os.umask(0o077); os.environ['CUDA_VISIBLE_DEVICES'] = ''

def sha(raw): return hashlib.sha256(raw).hexdigest()
def stamp(): return datetime.now(timezone.utc).isoformat()

def require(condition, code):
    if not condition: raise RuntimeError(code)

def load_metadata_operator():
    path = ROOT/'.runtime/export_matched_v11_operator.py'
    require(not path.is_symlink() and sha(path.read_bytes()) == EXPORT_OPERATOR_SHA, 'trusted_export_operator_source_required')
    spec = importlib.util.spec_from_file_location('backup_completed_metadata_operator', path)
    module = importlib.util.module_from_spec(spec); spec.loader.exec_module(module)
    return module

def historical_proof():
    """Read completed, original WRITE evidence. Never invoke exporter/live guards."""
    op = load_metadata_operator(); loaded = op.tools(); schema = loaded[1][0]
    pair = op.PAIR; directory = pair/'metadata-export-operators/metadata-export-v1'
    tracked = {}
    def read(path, maximum=8*1024**2):
        raw = op.read(path, maximum); tracked[str(path)] = sha(raw); return raw
    # Pair cleanup observations legitimately exceed the old private_read 64KiB default.
    pair_raw = read(pair/'controller-journal.json', PAIR_JOURNAL_LIMIT)
    journal_raw = read(directory/'journal.json'); journal = op.load(journal_raw)
    plan_raw = read(directory/'plan.json'); plan = op.load(plan_raw)
    manifest_raw = read(directory/'transport-manifest.json'); manifest = op.load(manifest_raw)
    marker = op.load(read(directory/'execution-marker.json'))
    require(journal.get('phase') == 'complete' and not journal.get('pending_action')
        and journal.get('transport_file_count') == 14 and journal.get('automatic_replay') is False
        and journal.get('business_state_modified') is False, 'completed_metadata_operator_required')
    for doc in (journal, plan, manifest):
        require(doc.get('version') == 1 and doc.get('operation') == op.OPERATION
            and doc.get('prepared_pair_sha256') == op.PLAN_SHA
            and doc.get('export_operation') == 'metadata-export-v1', 'historical_operator_binding_differs')
    prepared_sha = sha(plan_raw)
    require(plan.get('phase') == 'prepared' and plan.get('exports_root') == str(pair/'exports')
        and plan.get('operator_directory') == str(directory) and plan.get('automatic_replay') is False
        and journal.get('operator_prepared_sha256') == prepared_sha
        and manifest.get('operator_prepared_sha256') == prepared_sha
        and marker == {'phase':'execute_intent','operator_prepared_sha256':prepared_sha}, 'historical_prepared_binding_differs')
    source = op.source_guard(plan.get('source_sha256'))
    require(source.get('operator_source_sha256') == EXPORT_OPERATOR_SHA
        and journal.get('source_sha256') == source
        and manifest.get('operator_source_sha256') == EXPORT_OPERATOR_SHA, 'historical_operator_source_differs')
    receipts = {label:op.receipt(read(directory/'receipts'/(label+'-export-receipt.json')), label, True, schema)
        for label in op.JOBS}
    binding, validated_pair, comparison = op.context(receipts, loaded)
    require(binding == plan.get('binding') and validated_pair == pair_raw, 'historical_terminal_context_differs')
    require(read(pair/'comparison.json') == comparison, 'historical_comparison_original_differs')
    require(manifest.get('phase') == 'metadata_export_complete'
        and manifest.get('original_pair_journal_sha256') == sha(pair_raw)
        and manifest.get('comparison_sha256') == sha(comparison)
        and manifest.get('raw_outputs_or_env_or_dump_included') is False
        and manifest.get('business_state_modified') is False
        and journal.get('transport_manifest_sha256') == sha(manifest_raw), 'historical_transport_manifest_differs')
    expected = {}
    export_sha = {}
    for label, job in op.JOBS.items():
        target = pair/'exports'/label
        require(target.is_dir() and not target.is_symlink()
            and {p.name for p in target.iterdir()} == {job+'-'+suffix for suffix in op.SOURCE_NAMES}, 'exact_five_metadata_exports_required')
        for suffix in op.SOURCE_NAMES:
            relative = label+'/'+job+'-'+suffix
            expected[relative] = target/(job+'-'+suffix)
            export_sha[relative] = receipts[label]['source_inputs_sha256'][suffix]
        expected[label+'-export-receipt.json'] = directory/'receipts'/(label+'-export-receipt.json')
    expected.update({name:directory/'payload'/name for name in ('comparison.json','pair-controller-journal.json')})
    items = manifest.get('files')
    require(type(items) is list and len(items) == 14 and all(type(x) is dict and set(x) == {'relative_path','source_path','sha256','bytes'} for x in items), 'exact_fourteen_transport_files_required')
    require(len({x['relative_path'] for x in items}) == 14 and {x['relative_path'] for x in items} == set(expected), 'exact_fourteen_transport_paths_required')
    for item in items:
        path = expected[item['relative_path']]
        require(item['source_path'] == str(path), 'historical_transport_source_path_differs')
        raw = read(path)
        require(type(item['bytes']) is int and len(raw) == item['bytes'] and sha(raw) == item['sha256'], 'historical_transport_file_differs')
        if item['relative_path'] in export_sha:
            require(sha(raw) == export_sha[item['relative_path']], 'original_write_receipt_export_differs')
    require(read(directory/'payload/pair-controller-journal.json',PAIR_JOURNAL_LIMIT) == pair_raw
        and read(directory/'payload/comparison.json') == comparison, 'historical_payload_copy_differs')
    # Also bind original source artifacts used by context() throughout the backup.
    for label, job in op.JOBS.items():
        for suffix, filename in op.SOURCE_NAMES.items():
            require(sha(read(op.RUNS/job/filename)) == receipts[label]['source_inputs_sha256'][suffix], 'historical_source_artifact_drift')
    return {'matched_export_sha256':export_sha, 'matched_plan_sha256':op.PLAN_SHA,
        'metadata_export_operator_source_sha256':EXPORT_OPERATOR_SHA,
        'metadata_export_transport_manifest_sha256':sha(manifest_raw),
        'metadata_export_transport_file_count':14, 'historical_proof_file_sha256':tracked,
        'metadata_export_source_sha256':source}, op

def parse_args(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--operation', choices=OPERATIONS, default=OPERATIONS[0])
    return parser.parse_args(argv)

def main(argv=None):
    args = parse_args(argv)
    RUN = ROOT/'.runtime'/args.operation
    import psycopg
    from backend.migrate import MIGRATIONS
    from devtools.backup_postgres import select_bin_dir, preflight, create_backup
    from ops.supervise import load_manifest, private_read, private_write
    from ops.ha.set_scheduler_policy import owned_identity
    from ops.ha.deploy_storage_admission import bound_owner_dsn, ledger_invariants
    from ops.ha.finalize_api_policy import values
    if (RUN/'journal.json').exists(): raise RuntimeError('existing_job_no_resubmit')
    manifest, services, _ = load_manifest(ROOT/'.runtime/services-primary.json')
    watchdog = dict(role='supervisor', name='primary', env_file=str(ROOT.parent/'supervisor.env'))
    services = [*services, watchdog]
    identities = {s['role']+'-'+s['name']: owned_identity(ROOT,s) for s in services}
    files = {str(p): sha(private_read(p)) for p in [manifest,*[Path(s['env_file']) for s in services]]}
    expected = [(v,checksum) for v,_,checksum in MIGRATIONS]
    if expected[-1][0] != 11: raise RuntimeError('requires_actual_registry11')
    historical, export_operator = historical_proof()
    def source_state():
        result = {}
        for base in ('backend','worker','ops','scripts'):
            for path in sorted((ROOT/base).rglob('*')):
                if path.suffix not in ('.py','.sql'): continue
                if path.is_symlink(): raise RuntimeError('source_symlink_refused')
                if path.is_file(): result[str(path.relative_to(ROOT))]=sha(path.read_bytes())
        return result
    source = source_state()
    def guard():
        export_operator.source_guard(historical['metadata_export_source_sha256'])
        if any(sha(export_operator.read(Path(path),16*1024**2)) != checksum for path,checksum in historical['historical_proof_file_sha256'].items()):
            raise RuntimeError('historical_metadata_proof_drift')
        if source_state() != source: raise RuntimeError('protected_source_drift')
        if {s['role']+'-'+s['name']: owned_identity(ROOT,s) for s in services} != identities:
            raise RuntimeError('service_identity_drift')
        if any(sha(private_read(Path(p))) != checksum for p,checksum in files.items()):
            raise RuntimeError('protected_configuration_drift')
    apis = [values(private_read(Path(s['env_file'])))['DATABASE_URL'] for s in services if s['role']=='api']
    owner = bound_owner_dsn(private_read(ROOT.parent/'neon-owner.env'), apis)
    with psycopg.connect(owner,connect_timeout=10,options='-c default_transaction_read_only=on -c statement_timeout=10000') as c:
        ledger = c.execute('SELECT version,checksum FROM batchforge_schema_migrations ORDER BY version').fetchall()
        active = c.execute("SELECT count(*) FROM tasks WHERE status IN ('pending','running')").fetchone()[0]
        if ledger != expected or active: raise RuntimeError('exact_schema11_drained_required')
        invariants = ledger_invariants(c)
    guard()
    # The launcher keeps its log outside RUN. Exclusive mkdir claims this job.
    RUN.mkdir(mode=0o700)
    directory_fd = os.open(RUN.parent, os.O_RDONLY | os.O_DIRECTORY)
    try: os.fsync(directory_fd)
    finally: os.close(directory_fd)
    doc = dict(job_id=args.operation,phase='backing_up',started_at=stamp(),
        source_sha256=sha(Path(__file__).read_bytes()),owned_identities=identities,protected_file_sha256=files,
        schema_version=11,migration_checksums=dict(ledger),invariants_before=invariants,
        matched_operation='qwen235b-v11-matched-v1',**historical,protected_source_sha256=source,
        off_pod_copy_verified=False,off_pod_permission_status='pending_explicit_user_authorization',
        limitations=['Same-Pod SQL backup and restore; no export, provider DR, object backup or role/ACL recovery.'])
    private_write(RUN/'journal.json',doc)
    try:
        bins = select_bin_dir(ROOT.parent/'postgres-client-17/bin',dict(os.environ))
        metadata = preflight(owner,bins,dict(os.environ))
        snapshot, backup = create_backup(owner,bins,ROOT/'.runtime/backups',dict(os.environ),metadata,RUN/'fresh-backup-raw.json')
        guard(); doc.update(phase='restoring',snapshot_sha256=backup['snapshot_sha256'],snapshot_bytes=backup['snapshot_bytes'])
        private_write(RUN/'journal.json',doc)
        argv = [sys.executable,'-m','ops.ha.local_pg17_restore','--backup-report',str(RUN/'fresh-backup-raw.json'),
            '--server-root',str(ROOT.parent/'postgres-server-17/rootfs'),'--client-root',str(ROOT.parent/'postgres-client-17/rootfs'),
            '--report',str(RUN/'restore.json')]
        with (RUN/'restore.log').open('xb') as out:
            result = subprocess.run(argv,cwd=ROOT,stdin=subprocess.DEVNULL,stdout=out,stderr=subprocess.STDOUT,timeout=600)
        report = json.loads(private_read(RUN/'restore.json',2*1024**2))
        guard()
        if result.returncode or not report.get('passed') or report.get('schema_version') != 11 or not report.get('temporary_cluster_removed'):
            raise RuntimeError('same_pod_restore_failed')
        doc.update(phase='complete',passed=True,finished_at=stamp(),restore_report_sha256=sha(private_read(RUN/'restore.json',2*1024**2)),
            services_and_configuration_unchanged=True)
    except BaseException as e:
        doc.update(phase='failed_or_uncertain',passed=False,error_type=type(e).__name__,finished_at=stamp())
        private_write(RUN/'journal.json',doc); raise
    private_write(RUN/'journal.json',doc)
    print(json.dumps({k:doc[k] for k in ('phase','passed','snapshot_bytes','snapshot_sha256','restore_report_sha256','off_pod_copy_verified')}),flush=True)

if __name__ == '__main__':
    try: main()
    except BaseException as e:
        print(json.dumps({'failed':True,'error_type':type(e).__name__}),flush=True);raise SystemExit(1) from None
