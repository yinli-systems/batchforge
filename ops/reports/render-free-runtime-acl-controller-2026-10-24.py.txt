class Refusal(ValueError): pass


def require(value, code):
    if not value:
        raise Refusal(code)


def sha(raw): return hashlib.sha256(raw).hexdigest()
def stamp(): return datetime.now(timezone.utc).isoformat()
def encoded(value): return (json.dumps(value, sort_keys=True, separators=(',', ':'))+'\n').encode()


def source_guard():
    require(tuple(v for v, _, _ in MIGRATIONS) == tuple(range(1, 12))
            and tuple(d for _, _, d in MIGRATIONS) == CHECKSUMS
            and all(sha(statement.encode()) == checksum for _, statement, checksum in MIGRATIONS),
            'exact_immutable_eleven_migrations_required')
    grants = (ROOT/'ops/ha/runtime_grants.sql').read_bytes()
    require(sha(grants) == GRANTS_SHA and sha((ROOT/'ops/ha/local_pg17_restore.py').read_bytes()) == VERIFY_SHA,
            'exact_grants_and_verifier_source_required')
    return {'migration_checksums': list(CHECKSUMS), 'runtime_grants_sha256': GRANTS_SHA,
            'verifier_sha256': VERIFY_SHA}, grants.decode()


def preflight(conn):
    row = conn.execute("""SELECT current_database(),current_user,session_user,
      pg_get_userbyid(d.datdba)=current_user,r.rolsuper,r.rolcreaterole,
      current_setting('default_transaction_read_only'),current_setting('server_version_num')::integer,
      pg_database_size(d.oid) FROM pg_database d JOIN pg_roles r ON r.rolname=current_user
      WHERE d.datname=current_database()""").fetchone()
    require(row and NAME.fullmatch(row[0]) and row[1] == row[2] and row[3]
            and row[4] is False and row[5] is True and row[6] == 'on'
            and row[7] >= 160000 and row[8] <= 64*1024**2,
            'empty_owned_disposable_nonsuperuser_createrole_readonly_database_required')
    require(conn.execute(EMPTY_SQL).fetchone() == (0, 0, False),
            'all_user_namespaces_empty_and_runtime_role_absent_required')
    return {'database': row[0], 'owner': row[1], 'server_version_num': row[7],
            'database_bytes_before': row[8], 'nonsuperuser': True,
            'runtime_role_absent': True, 'user_relations_functions_before': [0, 0]}


class Probe:
    def __init__(self, conn, *, clock=time.monotonic):
        self.conn, self.clock, self.started, self.count = conn, clock, clock(), 0
        self.denials = []
        self.phase = 'begin'

    def reserve(self):
        require(self.count < 256 and self.clock()-self.started <= 120, 'finite_statement_or_time_budget_reached')
        self.count += 1

    def execute(self, statement, params=()):
        self.reserve()
        return self.conn.execute(statement, params) if params else self.conn.execute(statement)

    def deny(self, label, statement, params=(), state='42501'):
        try:
            with self.conn.transaction():
                self.execute(statement, params)
        except psycopg.Error as error:
            require(error.sqlstate == state, 'unexpected_denial_sqlstate')
            self.denials.append({'label': label, 'sqlstate': state})
        else:
            raise Refusal('forbidden_operation_unexpectedly_succeeded')


class VerifierConnection:
    """Meter canonical verifier execute calls without changing its queries."""
    def __init__(self, probe): self.probe = probe
    def execute(self, statement, params=()): return self.probe.execute(statement, params)
    def cursor(self, *args, **kwargs):
        probe = self.probe
        class Cursor:
            def __init__(self): self.actual = probe.conn.cursor(*args, **kwargs)
            def __enter__(self): self.actual.__enter__(); return self
            def __exit__(self, *values): return self.actual.__exit__(*values)
            def execute(self, statement, params=()):
                probe.reserve()
                return self.actual.execute(statement, params) if params else self.actual.execute(statement)
            def __iter__(self): return iter(self.actual)
        return Cursor()

def private_read(path, maximum=131072):
    path = Path(path); info = path.lstat()
    require(stat.S_ISREG(info.st_mode) and not path.is_symlink() and info.st_uid == os.geteuid()
            and not info.st_mode & 0o077 and info.st_nlink == 1 and 0 < info.st_size <= maximum, 'private_file_required')
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
    try:
        actual = os.fstat(fd); require((actual.st_dev, actual.st_ino, actual.st_size, actual.st_mtime_ns) ==
                                      (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns), 'private_file_changed')
        raw = os.read(fd, maximum+1); require(len(raw) == info.st_size, 'private_file_size_changed'); return raw
    finally: os.close(fd)


def load_dsn(path, host, database):
    require(NAME.fullmatch(database) and re.fullmatch(r'[A-Za-z0-9.-]{1,253}', host), 'explicit_target_required')
    raw = private_read(path); values = []
    for line in raw.decode().splitlines():
        if line.startswith('DATABASE_URL='):
            fields = shlex.split(line[13:], comments=False); require(len(fields) == 1, 'one_literal_dsn_required'); values.extend(fields)
    require(len(values) == 1, 'one_database_url_required')
    parts = conninfo_to_dict(values[0])
    require(parts.get('host') == host and parts.get('dbname') == database and parts.get('user') == 'batchforge_owner'
            and parts.get('sslmode') == 'verify-full' and not any(parts.get(k) for k in ('hostaddr', 'service')),
            'exact_strict_tls_qualification_target_required')
    port = parts.get('port') or '5432'
    require(re.fullmatch('[0-9]{1,5}', port) and 1 <= int(port) <= 65535, 'single_explicit_port_required')
    return make_conninfo(values[0], port=port), sha(raw)


def save(path, value, *, fresh=False):
    path = Path(path); raw = encoded(value)
    temporary = path if fresh else path.with_name(path.name+'.pending')
    fd = os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(fd, 'wb') as out: out.write(raw); out.flush(); os.fsync(out.fileno())
    if not fresh: os.replace(temporary, path)
    fd = os.open(path.parent, os.O_RDONLY | os.O_DIRECTORY)
    try: os.fsync(fd)
    finally: os.close(fd)


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ('access-env', 'run-directory'): parser.add_argument('--'+name, type=Path, required=True)
    for name in ('expected-host', 'expected-database'): parser.add_argument('--'+name, required=True)
    parser.add_argument('--execute', action='store_true'); parser.add_argument('--prepared-sha256')
    args = parser.parse_args(argv); os.umask(0o077)
    try:
        require(bool(args.prepared_sha256) == args.execute, 'separate_prepared_execution_required')
        source, _ = source_guard(); dsn, config_sha = load_dsn(args.access_env, args.expected_host, args.expected_database)
        # This is a separate operator process. Do not inherit a different
        # PGHOSTADDR/PGPORT/PGSERVICE/password/certificate from its caller.
        for key in tuple(os.environ):
            if key.startswith('PG'): os.environ.pop(key)
        run = args.run_directory
        if not args.execute: run.mkdir(mode=0o700)
        info = run.lstat(); require(stat.S_ISDIR(info.st_mode) and not run.is_symlink()
            and info.st_uid == os.geteuid() and not info.st_mode & 0o077, 'private_new_run_directory_required')
        with psycopg.connect(dsn, autocommit=True, connect_timeout=5,
             options='-c default_transaction_read_only=on -c statement_timeout=5000 -c lock_timeout=1000 -c idle_in_transaction_session_timeout=15000') as conn:
            require(conn.pgconn.ssl_in_use, 'actual_tls_required')
            observed = preflight(conn)
            binding = {'target': observed, 'configuration_sha256': config_sha, 'source_binding': source,
                       'operator_source_sha256': sha(Path(__file__).read_bytes())}
            prepared = {'phase': 'prepared', 'binding': binding, 'automatic_replay': False}
            if not args.execute:
                save(run/'prepared.json', prepared, fresh=True)
                print(json.dumps({'phase': 'prepared', 'prepared_sha256': sha(private_read(run/'prepared.json'))})); return 0
            raw = private_read(run/'prepared.json'); require(sha(raw) == args.prepared_sha256 and json.loads(raw) == prepared, 'exact_prepared_binding_required')
            require(not (run/'report.json').exists() and not (run/'execution.claim').exists(), 'existing_execution_no_replay')
            save(run/'execution.claim', {'prepared_sha256': args.prepared_sha256, 'nonce': secrets.token_hex(16), 'automatic_replay': False}, fresh=True)
            save(run/'report.json', {'phase': 'started_or_unknown', 'passed': False, 'pending_action': 'rollback_only_qualification_transaction', 'prepared_sha256': args.prepared_sha256}, fresh=True)
            result = qualify(conn, observed)
            report = {'phase': 'complete' if result['passed'] else 'failed_or_unknown', 'pending_action': None if result['rollback_confirmed'] else 'rollback_ack_unknown', 'prepared_sha256': args.prepared_sha256,
                      'at': stamp(), 'automatic_replay': False, **result}
            save(run/'report.json', report)
            print(json.dumps({k: report[k] for k in ('phase', 'passed', 'rollback_confirmed', 'pending_action', 'statement_calls')})); return 0 if report['passed'] else 2
    except BaseException:
        print('{"phase":"refused_or_outcome_unknown","passed":false,"automatic_replay":false}'); return 2
