def json_safe_default(value):
 if isinstance(value,Decimal):
  if not value.is_finite():raise ValueError('nonfinite_decimal_evidence')
  if value==value.to_integral_value():return int(value)
  number=float(value)
  if not math.isfinite(number):raise ValueError('decimal_evidence_overflow')
  return number
 if isinstance(value,(datetime.datetime,datetime.date)):return value.isoformat()
 raise TypeError('unsupported_evidence_type_'+type(value).__name__)

class WorkerObserver:
 def __init__(self):self.samples=[];self.done=threading.Event();self.thread=threading.Thread(target=self.observe,daemon=True)
 def start(self):self.thread.start()
 def observe(self):
  while not self.done.is_set():
   try:self.samples.append(worker_metrics())
   except Exception as e:self.samples.append({'at':stamp(),'error_type':type(e).__name__})
   if len(self.samples)>=7200:break
   self.done.wait(1)
 def stop(self):self.done.set();self.thread.join(timeout=5)
 def summary(self):
  valid=[s for s in self.samples if 'gauges'in s]
  if len(valid)<1:raise RuntimeError('worker_observation_insufficient')
  for sample in valid:validate_worker_sample(sample)
  return {'valid_samples':len(valid),'error_samples':len(self.samples)-len(valid),'sampled_maxima':{k:max(s['gauges'][k] for s in valid) for k in WORKER_GAUGES},'scope':'single worker logical exact-token reservations, not physical GPU KV occupancy; inflight includes tokenization, reserve waiters, inference and reporting; no stage counts inferred by subtraction','peak_scope':'adaptive_kv_peak_tokens is process-lifetime cumulative; reserved_context_tokens sampled maximum is the observed workload peak, not a guaranteed continuous maximum'}

def archive_state(env,bid):
 from psycopg.rows import dict_row

  conn.execute('SET TRANSACTION ISOLATION LEVEL REPEATABLE READ')

  attempts=conn.execute("SELECT count(*) AS n,count(*) FILTER(WHERE status='completed') AS accepted,COALESCE(sum(prompt_tokens) FILTER(WHERE status='completed'),0) AS prompt_tokens,COALESCE(sum(completion_tokens) FILTER(WHERE status='completed'),0) AS completion_tokens FROM attempts WHERE batch_id=%s",(bid,)).fetchone()

   files=conn.execute('SELECT file_id,kind,bytes,line_count,expected_count,sha256 FROM result_archive_files WHERE archive_id=%s',(r['id'],)).fetchall()
