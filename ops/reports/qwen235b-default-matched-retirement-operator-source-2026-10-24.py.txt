def load(raw):
    def unique(pairs):
        result={}
        for k,v in pairs:require(k not in result,'duplicate_metadata_key');result[k]=v

    def invalid(_):raise Refusal('nonfinite_metadata')
    return json.loads(raw.decode('utf-8'),object_pairs_hook=unique,parse_constant=invalid)

def digest_query(conn,query,args=()):
    """Only per-row hashes leave PostgreSQL, never body/error/answer/key text."""
    result=conn.execute("SELECT encode(sha256(convert_to(row_to_json(q)::text,'UTF8')),'hex') FROM ("+query+') q ORDER BY 1',args)
    digest=hashlib.sha256();count=0
    for (value,) in result:
        digest.update(value.encode()+b'\n');count+=1
    return dict(rows=count,sha256=digest.hexdigest())

def refund_arithmetic(before,usage,retired):
    rows_before={b['batch_id']:b for b in before['selected']}
    require(len(set(retired))==len(retired) and set(retired)<=rows_before.keys(),'invalid_confirmed_batch_set')
    refunded_tasks=sum(rows_before[bid]['tasks'] for bid in retired);refunded_bytes=sum(rows_before[bid]['charged_bytes'] for bid in retired)
    require(usage['retained_tasks']==before['usage']['retained_tasks']-refunded_tasks
        and usage['charged_payload_bytes']==before['usage']['charged_payload_bytes']-refunded_bytes,'measured_retention_refund_differs')
    return dict(refunded_tasks=refunded_tasks,refunded_payload_bytes=refunded_bytes,
        retired_attempts=sum(rows_before[bid]['attempts'] for bid in retired))


def compare_snapshots(before,after,retired):
    require(before['protection']==after['protection'] and before['legacy']==after['legacy'],'outside_history_output_or_usage_changed')
    require(not any(after['invariants'].values()),'final_ledger_invariant_failed')
    return refund_arithmetic(before,after['usage'],retired)


def policy_cas(conn,previous,enabled):
    """Same-TX prior-state CAS. No outer lock followed by a second helper connection."""
    target=target_policy(previous,enabled)
    with conn.transaction():
        conn.execute('SET TRANSACTION ISOLATION LEVEL READ COMMITTED')
        require(conn.execute("SELECT id FROM tenants WHERE id='default' FOR NO KEY UPDATE NOWAIT").fetchone()==(TENANT,),'default_tenant_missing')
        actual=policy(conn,locked=True);require(actual==previous,'policy_prior_state_changed')
        if enabled:
            require(previous is None or previous['enabled'] is False and previous['replay_ttl_seconds']==TTL,'policy_already_enabled_or_ttl_differs')
            if previous is None:
                conn.execute('INSERT INTO tenant_history_retirement_policy(tenant_id,enabled,eligible_created_after,min_age_seconds,replay_ttl_seconds) VALUES(%s,true,%s,%s,%s)',(TENANT,CUTOFF,MIN_AGE,TTL))
            else:
                conn.execute("UPDATE tenant_history_retirement_policy SET enabled=true,eligible_created_after=%s,min_age_seconds=%s,replay_ttl_seconds=%s WHERE tenant_id='default'",(CUTOFF,MIN_AGE,TTL))
        else:
            require(previous==target_policy(previous,True),'disable_requires_exact_owned_enabled_policy')
            conn.execute("UPDATE tenant_history_retirement_policy SET enabled=false WHERE tenant_id='default'")
        result=policy(conn);require(result==target,'policy_target_or_floor_changed')
    return result  # Only after the COMMIT acknowledgement.


async def verify_archive(pool,meter,item,expected=None):
    verifier=ResultArchiver(pool,meter,max_archived_bytes_per_tenant=1)
    async with pool.connection() as conn:
        run,files,chunks=await verifier._ready(conn,TENANT,item['batch_id'])
    require(run and run['purged_at'] is not None and run['writer_token'] is None and run['lease_expires_at'] is None,'ready_purged_writerless_archive_required')
    require(len(files)==1 and files[0]['file_id']==item['output_file_id'] and files[0]['kind']=='output'
        and files[0]['bytes']==item['output_bytes'] and files[0]['sha256']==item['output_sha256']
        and files[0]['line_count']==files[0]['expected_count']==PAGE,'collected_archive_file_identity_differs')
    require(1<=len(chunks)<=PAGE and all(type(c['bytes']) is int and c['bytes']>0 for c in chunks),'bounded_positive_archive_chunks_required')
    declared=sum(c['bytes'] for c in chunks)
    require(declared==item['output_bytes'] and declared*3<=BYTE_LIMIT,'three_verification_passes_exceed_page_budget')
    identity=clean(dict(archive_id=run['id'],manifest_sha256=run['manifest_sha256'],file_id=files[0]['file_id'],
        bytes=files[0]['bytes'],sha256=files[0]['sha256'],count=files[0]['line_count']))
    if expected is not None:require(identity==expected,'archive_generation_changed')
    await asyncio.to_thread(verifier._verify_files,files,chunks)
    return identity


def exact_page(result,expected,attempts,apply):
    require(isinstance(result,dict) and set(result)=={'eligible','apply','tasks','attempts','remaining_tasks','archive_id','manifest_sha256'}
        and result['eligible'] is True and result['apply'] is apply
        and all(type(result[k]) is int for k in ('tasks','attempts','remaining_tasks'))
        and result['tasks']==PAGE and result['attempts']==attempts and PAGE<=attempts<=PAGE*3
        and result['remaining_tasks']==(0 if apply else PAGE)
        and result['archive_id']==expected['archive_id'] and result['manifest_sha256']==expected['manifest_sha256'],
        'exact_page_outcome_or_attempt_count_differs')
    return result


class Driver:
    """Small injected state machine; Context does all private production observation."""
    def __init__(self,context,journal,prepared,*,clock=time.monotonic,sleep=asyncio.sleep,stop=lambda:False):
        require(1<=len(context.batches)<=MAX_PAGES and len({b['batch_id'] for b in context.batches})==len(context.batches)
            and all(b['tasks']==PAGE for b in context.batches),'bounded_unique_twenty_task_batches_required')
        self.context=context;self.journal=journal;self.report=prepared;self.clock=clock;self.sleep=sleep;self.stop=stop;self.started=False
    def save(self,value):self.journal.save(clean(value));self.report=clean(value)
    async def action(self,stage,callback,**metadata):
        self.context.guard();require(not self.stop(),'interrupted_before_action')
        require(not self.report.get('pending_action'),'unresolved_action_no_replay')
        candidate={**self.report,'phase':stage+'_intent','pending_action':dict(stage=stage,**metadata)}
        self.journal.append('intent',stage=stage,**metadata);self.save(candidate)
        result=await callback()
        candidate={**self.report,'phase':stage+'_confirmed','pending_action':None}
        candidate.setdefault('actions',[])
        # Copy rather than mutate the report that still contains the durable intent.
        candidate['actions']=[*candidate['actions'],dict(stage=stage,result=clean(result),**metadata)]
        self.journal.append('confirmed',stage=stage,**metadata);self.save(candidate)
        return result
    async def run(self):
        require(not self.started and self.report.get('phase')=='prepared' and not self.report.get('pending_action'),'operation_already_started_no_replay')
        self.started=True;retired=[];started=self.clock()
        try:
            self.context.guard()
            before=getattr(self.context,'initial_snapshot',None) or await self.context.snapshot(retired)
            require(before==self.report['before_snapshot'],'prepared_snapshot_changed')
            prior=before['policy'];require(prior is None or prior['enabled'] is False and prior['replay_ttl_seconds']==TTL,'default_policy_not_disabled')
            archives={}
            for item in self.context.batches:
                self.context.guard();require(not self.stop(),'interrupted_during_readonly_verification')
                archives[item['batch_id']]=await self.context.archive(item)
            self.save({**self.report,'archives_before':archives,'phase':'archives_verified'})
            async def enable():return self.context.policy(prior,True)
            enabled=await self.action('policy_enable',enable)
            # Preview EVERY exact batch before deleting even the first page.
            for item in self.context.batches:
                self.context.guard();require(not self.stop(),'interrupted_before_preview')
                preview=await self.context.preview(item)
                expected=archives[item['batch_id']]
                attempts=next(b['attempts'] for b in before['selected'] if b['batch_id']==item['batch_id'])
                exact_page(preview,expected,attempts,False)
            last=None
            for item in self.context.batches:
                if last is not None:
                    wait=max(0,10-(self.clock()-last))
                    if wait:await self.sleep(wait)
                self.context.guard();require(not self.stop(),'interrupted_before_page')
                current=await self.context.progress(before,retired)
                require(current['policy']==enabled,'enabled_policy_drift')
                expected=archives[item['batch_id']]
                attempts=next(b['attempts'] for b in before['selected'] if b['batch_id']==item['batch_id'])
                async def page(item=item,expected=expected,attempts=attempts):
                    result=await self.context.page(item)
                    return exact_page(result,expected,attempts,True)
                result=await self.action('history_page',page,batch_id=item['batch_id'],limit=PAGE,manifest_sha256=expected['manifest_sha256'])
                retired.append(item['batch_id']);last=self.clock()
                current=await self.context.progress(before,retired);refund=current['measured_refund']
                self.save({**self.report,'confirmed_batches':retired[:],'measured_refund':refund})
            after=await self.context.snapshot(retired);refund=compare_snapshots(before,after,retired)
            require(after['policy']==enabled,'policy_changed_before_final_verify')
            final={}
            for item in self.context.batches:
                self.context.guard();require(not self.stop(),'interrupted_during_final_verify')
                final[item['batch_id']]=await self.context.archive(item,archives[item['batch_id']])
            self.save({**self.report,'archives_after':final,'after_snapshot':after,'measured_refund':refund})
            async def disable():return self.context.policy(enabled,False)
            disabled=await self.action('policy_disable',disable)
            end=await self.context.snapshot(retired);compare_snapshots(before,end,retired)
            require(end['policy']==disabled,'final_disabled_policy_changed')
            self.context.guard()
            self.save({**self.report,'phase':'complete','passed':True,'pending_action':None,'finished_at':stamp(),
                'elapsed_seconds':self.clock()-started,'final_snapshot':end,'verified_archive_bytes':self.context.verified_bytes(),
                'retire_keys':False,'replay_floor_unchanged':True,'all_selected_output_hashes_unchanged':True})
            return self.report
        except BaseException as error:
            candidate={**self.report,'phase':'manual_recovery_required','passed':False,'manual_recovery_required':True,
                'error_type':type(error).__name__,'error_code':str(error) if type(error) is Refusal else type(error).__name__,
                'finished_at':stamp(),'automatic_replay':False}
            try:self.save(candidate)
            except BaseException:pass
            raise  # No finally disable, retry, cleanup, policy compensation or workload resubmission.

    def processes(self):

            for name in ('environ','cmdline'):

    def current_binding(self):
        source={}

            for p in sorted((ROOT/base).rglob('*')):
                if p.suffix in ('.py','.sql'):
                    require(not p.is_symlink(),'source_symlink_refused')

        source[str(SELF.relative_to(ROOT))]=sha(SELF.read_bytes())
        source.update({p:sha((ROOT/p).read_bytes()) for p in [*PINS,'.runtime/prepare_default_matched_retirement.py']})
        files={str(p):sha(private_file(p)) for p in [self.manifest,*[Path(s['env_file']) for s in self.services],self.args.owner_env,self.args.access_env]}
        return clean(dict(source_sha256=source,environment_sha256=files,service_identities=self.processes(),archive_identity_sha256=self.archive_identity,
            input_sha256=self.inputs,backup=self.backup))
    def guard(self):

            require(sha(read(self.operation_spec[0],32*1024**2))==self.operation_spec[1],'immutable_prepared_operation_changed')
        observed=datetime.fromisoformat(self.backup['observed_at'].replace('Z','+00:00'))

        for path,digest in self.backup['proof_files'].items():require(sha(read(Path(path),4*1024**2))==digest,'backup_proof_changed')
        archive=Path(self.backup['snapshot_path']);info=archive.lstat()
        require(not archive.is_symlink() and {k:getattr(info,k) for k in self.backup['snapshot_stat']}==self.backup['snapshot_stat'],'backup_snapshot_identity_changed')
        require(self.current_binding()==self.binding,'prepared_source_env_pid_input_or_backup_drift')
        require(all(sha(read(Path(p)))==h for p,h in self.inputs.items()),'source_metadata_changed')
    async def snapshot(self,retired):

        count=await accounting(self.pool,TENANT)
        require(value['usage']=={k:count[k] for k in ('retained_tasks','charged_payload_bytes')},'accounting_snapshot_changed')
        return value
    async def progress(self,before,retired):
        # Small per-page conservation check, not a repeated full-table/raw-row transport.
        count=await accounting(self.pool,TENANT)
        with self.owner.transaction():
            self.owner.execute('SET TRANSACTION ISOLATION LEVEL REPEATABLE READ READ ONLY')
            require(one(self.owner,"SELECT count(*) AS n FROM tasks WHERE status IN ('pending','running')")['n']==0,'global_drain_required')
            current_policy=policy(self.owner)
            current_ids=[r[0] for r in self.owner.execute("SELECT id FROM batches WHERE id=ANY(%s) AND history_retired_tasks=20 AND history_retired_at IS NOT NULL ORDER BY id",([b['batch_id'] for b in self.batches],))]
            require(current_ids==sorted(retired),'confirmed_page_markers_differ')
        refund=refund_arithmetic(before,{k:count[k] for k in ('retained_tasks','charged_payload_bytes')},retired)
        return dict(policy=current_policy,measured_refund=refund)
    async def archive(self,item,expected=None):return await verify_archive(self.pool,self.meters[item['batch_id']],item,expected)
    async def preview(self,item):return await history_page(self.pool,TENANT,item['batch_id'],limit=PAGE,apply=False)
    async def page(self,item):

    def policy(self,previous,enabled):return policy_cas(self.owner,previous,enabled)
    def verified_bytes(self):return sum(m.verified_bytes for m in self.meters.values())


@contextmanager
def named_lock(path):
    fd=os.open(path,os.O_RDWR|os.O_CREAT|os.O_NOFOLLOW,0o600)
    try:
        info=os.fstat(fd);require(stat.S_ISREG(info.st_mode) and info.st_uid==os.geteuid() and not info.st_mode&0o077 and info.st_nlink==1,'unsafe_operation_lock')
        fcntl.flock(fd,fcntl.LOCK_EX|fcntl.LOCK_NB);yield
    finally:os.close(fd)

def write_prepared(directory,report):
    path=directory/'prepared.json'
    fd=os.open(path,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
    with os.fdopen(fd,'wb') as out:out.write(encode(report));out.flush();os.fsync(out.fileno())
    fd=os.open(directory,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
    try:os.fsync(fd)
    finally:os.close(fd)
    return sha(read(path,32*1024**2))

def claim(directory,prepared_sha):
    path=directory/'execution.claim'
    fd=os.open(path,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
    with os.fdopen(fd,'wb') as out:out.write(encode(dict(prepared_sha256=prepared_sha,at=stamp(),automatic_replay=False)));out.flush();os.fsync(out.fileno())
    fd=os.open(directory,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
    try:os.fsync(fd)
    finally:os.close(fd)
