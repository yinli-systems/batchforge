def metrics():

  for sample in family.samples:
   if sample.name not in {family.name+'_sum',family.name+'_count',family.name+'_bucket'}:continue
   if not math.isfinite(sample.value):raise RuntimeError('nonfinite_metric')

   selected.append({'name':sample.name,'labels':sample.labels,'value':sample.value})
 if {s['name'].removesuffix('_count') for s in selected if s['name'].endswith('_count')}!=HISTOGRAMS:raise RuntimeError('missing_histogram')

def phase_delta(before,after,expected):
 def key(s):return (s['name'],tuple(sorted(s['labels'].items())))
 b={key(s):s['value'] for s in before['series']};a={key(s):s['value'] for s in after['series']}
 if set(a)!=set(b):raise RuntimeError('histogram_series_changed')
 d={k:a[k]-b[k] for k in a}
 if any(v < -1e-9 for v in d.values()):raise RuntimeError('histogram_counter_reset')
 out={}
 for family in sorted(HISTOGRAMS):
  count=sum(v for (name,_),v in d.items() if name==family+'_count');total=sum(v for (name,_),v in d.items() if name==family+'_sum')
  buckets={}
  for (name,labels),value in d.items():
   if name==family+'_bucket':
    le=float(dict(labels)['le']);buckets[le]=buckets.get(le,0)+value
  quantiles={}
  for q in (.5,.95,.99):
   eligible=[le for le,c in sorted(buckets.items()) if c>=count*q]
   upper=eligible[0] if eligible and count>0 else None
   quantiles[str(q)]=upper if upper is not None and math.isfinite(upper) else '+Inf' if upper is not None else None
  out[family]={'count':count,'sum_seconds':total,'mean_seconds':total/count if count else None,'percentile_bucket_upper_seconds':quantiles,'delta_buckets':{str(k):v for k,v in sorted(buckets.items())}}
 # Per-output-token observations are not per-request counts.
 for family in HISTOGRAMS-{'vllm:time_per_output_token_seconds'}:
  if out[family]['count']!=expected:raise RuntimeError('request_histogram_count_mismatch_'+family)
 return {'before_at':before['at'],'after_at':after['at'],'histograms':out,'percentile_method':'first cumulative bucket whose delta count >= quantile*delta_count; bucket upper bounds, not interpolated exact percentiles','limitations':['Prefill/decode means are engine wall-time stages under continuous batching, not CUDA kernel time.','Snapshot interval brackets workload submission and output validation. Exact expectedrequest observations and unchanged engine identity are required.']}

def worker_metrics():

def validate_worker_sample(sample,idle=False):

class WorkerObserver:
 def __init__(self):self.samples=[];self.done=threading.Event();self.thread=threading.Thread(target=self.observe,daemon=True)
 def start(self):self.thread.start()
 def observe(self):
  while not self.done.is_set():
   try:self.samples.append(worker_metrics())
   except Exception as e:self.samples.append({'at':stamp(),'error_type':type(e).__name__})
   if len(self.samples)>=7200:break
   self.done.wait(1)
 def stop(self):self.done.set();self.thread.join(timeout=5)
 def summary(self):
  valid=[s for s in self.samples if 'gauges'in s]
  if len(valid)<5:raise RuntimeError('worker_observation_insufficient')
  for sample in valid:validate_worker_sample(sample)
  return {'valid_samples':len(valid),'error_samples':len(self.samples)-len(valid),'sampled_maxima':{k:max(s['gauges'][k] for s in valid) for k in WORKER_GAUGES},'scope':'single worker logical exact-token reservations, not physical GPU KV occupancy; inflight includes tokenization, reserve waiters, inference and reporting; no stage counts inferred by subtraction','peak_scope':'adaptive_kv_peak_tokens is process-lifetime cumulative; reserved_context_tokens sampled maximum is the observed workload peak, not a guaranteed continuous maximum'}
