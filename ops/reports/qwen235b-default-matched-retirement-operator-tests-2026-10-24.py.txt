class MemoryJournal:
    def __init__(self,fail=None):self.events=[];self.values=[];self.fail=fail
    def append(self,event,**values):
        if self.fail==('append',event,values.get('stage')):raise OSError('synthetic fsync failure')
        self.events.append((event,copy.deepcopy(values)))
    def save(self,value):
        if self.fail==('save',value['phase']):raise OSError('synthetic fsync failure')
        self.values.append(copy.deepcopy(value))
class FakeContext:
    def __init__(self,batches=None):
        self.batches=batches or [item(0),item(1)];self.current_policy=None;self.retired=[];self.calls=[];self.failure=None
        self.clock=0.;self.before=snapshot(self.batches)
    def guard(self):
        if self.failure=='guard':raise M.Refusal('changed_source')
    async def snapshot(self,retired):
        self.calls.append(('snapshot',tuple(retired)))
        result=snapshot(self.batches,retired,self.current_policy)
        if self.failure=='final_outside' and retired:result['protection']['outside']['sha256']='b'*64
        return result
    async def progress(self,before,retired):
        result=await self.snapshot(retired);refund=M.compare_snapshots(before,result,retired)
        return dict(policy=self.current_policy,measured_refund=refund)
    async def archive(self,item,expected=None):
        self.calls.append(('archive',item['batch_id'],expected is not None))
        if self.failure=='archive':raise M.Refusal('corrupt_synthetic_archive')
        return dict(archive_id='archive-'+item['batch_id'],manifest_sha256='b'*64)
    def policy(self,previous,enabled):
        self.calls.append(('policy',enabled));assert previous==self.current_policy
        self.current_policy=M.target_policy(previous,enabled)
        if self.failure==('policy',enabled):raise OSError('unknown commit acknowledgement')
        return copy.deepcopy(self.current_policy)
    async def preview(self,item):
        self.calls.append(('preview',item['batch_id']))
        return dict(eligible=self.failure!='preview',apply=False,tasks=20,attempts=20,remaining_tasks=20,
            archive_id='archive-'+item['batch_id'],manifest_sha256='b'*64)
    async def page(self,item):
        self.calls.append(('page',item['batch_id'],self.clock));self.retired.append(item['batch_id'])
        if self.failure=='page':raise OSError('unknown committed page')
        return dict(eligible=True,apply=True,tasks=20,attempts=20,remaining_tasks=0,
            archive_id='archive-'+item['batch_id'],manifest_sha256='b'*64)
    def verified_bytes(self):return 600
    async def sleep(self,seconds):assert seconds>=0;self.clock+=seconds

def driver(context,journal=None,stop=lambda:False):
    journal=journal or MemoryJournal();prepared=dict(phase='prepared',before_snapshot=context.before,pending_action=None)
    return M.Driver(context,journal,prepared,clock=lambda:context.clock,sleep=context.sleep,stop=stop)

def test_immutable_prepared_spec_cannot_be_overwritten(tmp_path,monkeypatch):
    # Spec bytes live separately from the mutable execution summary.
    root=tmp_path.resolve();directory=root/'.runtime/operation';directory.mkdir(parents=True,mode=0o700)
    monkeypatch.setattr(M,'ROOT',root)
    spec={'phase':'prepared','operation':'SYNTHETIC','pending_action':None}
    digest=M.write_prepared(directory,spec)
    journal=M.Journal(directory);journal.save(spec);journal.save({**spec,'phase':'policy_enable_intent','pending_action':'policy_enable'})
    assert M.sha((directory/'prepared.json').read_bytes())==digest
    with pytest.raises(FileExistsError):M.write_prepared(directory,spec)
