"""Validate/export metadata for an exact terminal fresh schema11 matched round.

Same-Pod only: no SSH, model-output export, DB dump, mutation or inference.
Default only validates; --write creates new private metadata files. Requires the
operator-reviewed prepared-plan SHA, exact job/round and unchanged live runtime.
Past schema10 baseline/pilot reports cannot be selected or relabeled here.
"""

class ExportRefusal(RuntimeError):
    pass

def require(condition,code):
    if not condition: raise ExportRefusal(code)

def read(path,maximum=16*1024**2):
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        info=os.fstat(fd)
        require(stat.S_ISREG(info.st_mode) and info.st_uid==os.geteuid() and not info.st_mode&0o077 and 0<info.st_size<=maximum,'unsafe_private_metadata')
        with os.fdopen(fd,'rb',closefd=False) as handle: raw=handle.read(maximum+1)
        after=os.fstat(fd)
        require(len(raw)==info.st_size and all(getattr(info,k)==getattr(after,k) for k in ('st_ino','st_dev','st_size','st_mtime_ns','st_ctime_ns')),'metadata_changed')
        return raw
    finally: os.close(fd)

def sha(raw): return hashlib.sha256(raw).hexdigest()

def load(raw):
    def unique(pairs):
        result={}
        for key,value in pairs:
            require(key not in result,'duplicate_metadata_key')
            result[key]=value
        return result
    def constant(_): raise ExportRefusal('nonfinite_metadata')
    return json.loads(raw.decode('utf-8'),object_pairs_hook=unique,parse_constant=constant)

def safe_metadata(value):
    denied={'response','body','content','messages','choices','text','answer','reasoning','prompt','password','api_key','worker_token','database_url','dsn','attempt_token'}
    def visit(item,depth=0):
        require(depth<=128,'metadata_nesting_bound')
        if isinstance(item,dict):
            require(not any(str(k).lower() in denied for k in item),'raw_or_secret_field_refused')
            for v in item.values():visit(v,depth+1)
        elif isinstance(item,list):
            for v in item:visit(v,depth+1)
        elif isinstance(item,str):
            require(not re.search(r'(?i)(?:postgres(?:ql)?://|bearer\s+[A-Za-z0-9_-]{16}|(?:https?://)[^/\s]*@)',item),'credential_like_text_refused')
    visit(value)
    json.dumps(value,allow_nan=False,ensure_ascii=False).encode('utf-8')

def validate_schema(value,schema,depth=0):
    require(depth<=128,'metadata_nesting_bound')
    if isinstance(schema,dict):
        require(isinstance(value,dict),'metadata_schema_type')
        allowed={k.removeprefix('?'):v for k,v in schema.items()}
        required={k for k in schema if not k.startswith('?')}
        require(required<=value.keys() and value.keys()<=allowed.keys(),'unknown_or_missing_metadata_field')
        for key,item in value.items():validate_schema(item,allowed[key],depth+1)
    elif isinstance(schema,tuple):
        kind=schema[0]
        if kind=='nullable':
            if value is not None:validate_schema(value,schema[1],depth+1)
        elif kind=='list':
            require(isinstance(value,list) and len(value)<=50000,'metadata_schema_type')
            for item in value:validate_schema(item,schema[1],depth+1)
        elif kind=='tuple':
            require(isinstance(value,list) and len(value)==len(schema[1]),'metadata_schema_type')
            for item,sub in zip(value,schema[1]):validate_schema(item,sub,depth+1)
        elif kind=='map':
            require(isinstance(value,dict) and len(value)<=50000,'metadata_schema_type')
            for key,item in value.items():
                require(isinstance(key,str) and re.fullmatch(schema[1],key) is not None,'unknown_metadata_map_key')
                validate_schema(item,schema[2],depth+1)
        elif kind=='pattern':require(isinstance(value,str) and re.fullmatch(schema[1],value) is not None,'metadata_schema_pattern')
        else:raise ExportRefusal('internal_schema_invalid')
    elif schema==S:require(isinstance(value,str) and len(value)<=16384,'metadata_schema_type')
    elif schema==I:require(type(value) is int,'metadata_schema_type')
    elif schema==N:require(type(value) in (int,float) and (type(value) is int or __import__('math').isfinite(value)),'metadata_schema_type')
    elif schema==B:require(type(value) is bool,'metadata_schema_type')
    else:raise ExportRefusal('internal_schema_invalid')

def validate_artifact(name,raw):
    require(name in SCHEMAS,'unknown_metadata_artifact')
    value=load(raw);validate_schema(value,SCHEMAS[name]);safe_metadata(value)
    return value
