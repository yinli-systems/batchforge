@contextmanager
def locked(path):
    fd=os.open(path,os.O_RDWR|os.O_CREAT|os.O_NOFOLLOW,0o600)
    try:
        info=os.fstat(fd);require(stat.S_ISREG(info.st_mode) and info.st_uid==os.geteuid()
            and info.st_nlink==1 and not info.st_mode&0o022,'unsafe_role_lock')
        fcntl.flock(fd,fcntl.LOCK_EX|fcntl.LOCK_NB);yield
    finally:os.close(fd)
def api_stop(root,controller,service,expected,check):
    from ops.ha.deploy_upload_protocol import private_file
    require(service['role']=='api' and service['name'] in {'a','b'},'exact_api_stop_only')
    path=root/'.runtime'/('api-'+service['name']+'.json');began=time.monotonic()
    with locked(path.with_suffix('.lock')):
        require(controller.record(service)==expected,'api_identity_changed_before_pidfd')
        fd=os.pidfd_open(expected['pid'])
        try:
            require(controller.record(service)==expected,'api_identity_changed_after_pidfd')
            signal.pidfd_send_signal(fd,signal.SIGTERM);poll=select.poll();poll.register(fd,select.POLLIN)
            while not poll.poll(500):check();require(time.monotonic()-began<30,'api_graceful_stop_timeout')
            require(json.loads(private_file(path))==expected,'api_original_pidfile_changed');path.unlink()
            descriptor=os.open(path.parent,os.O_RDONLY|os.O_DIRECTORY)
            try:os.fsync(descriptor)
            finally:os.close(descriptor)
        finally:os.close(fd)

class Roll:
    def __init__(self,root,run,manifest,services,target,h,controller,*,process_reader=None,starter=None,
                 api_stopper=api_stop,drained=None,idle=None,sleep=time.sleep,clock=time.monotonic,stop=None):

        require(target in {'fair','cost_aware'},'supported_policy_required')
        require(len(services)==5 and {(s['role'],s['name']) for s in services}==

        self.apis=sorted((s for s in services if s['role']=='api'),key=lambda s:s['name']);self.api=self.apis[0]

        self.worker=next(s for s in services if s['role']=='worker');self.engine=next(s for s in services if s['role']=='engine')
        self.watchdog=dict(role='supervisor',name='primary',env_file=str(root.parent/'supervisor.env'),python=self.api['python'])
        self.all=[*services,self.watchdog];self.env={s['env_file']:h['private_file'](Path(s['env_file'])) for s in self.all}
        require(h['values'](self.env[self.watchdog['env_file']]).get('SUPERVISOR_MANIFEST')==str(manifest),'watchdog_manifest_differs')
        self.original=dict(self.env);self.manifest_raw=h['private_file'](manifest);self.reader=process_reader or h['process_data']
        from ops.ha.set_scheduler_policy import policy_bytes
        self.patched={s['env_file']:policy_bytes(self.env[s['env_file']],target) for s in self.apis}
        old=[h['values'](self.env[s['env_file']]).get('SCHEDULER_POLICY','legacy') for s in self.apis]
        require(old[0]==old[1] and old[0] in {'fair','cost_aware'} and old[0]!=target,'matching_other_original_policy_required')
        self.worker_values=h['values'](self.env[self.worker['env_file']]);self.profile_value=json.loads(self.worker_values['SCHEDULER_PROFILE_JSON'])
        self.budget=int(self.worker_values['WORKER_TOKEN_BUDGET'])
        engine=h['values'](self.env[self.engine['env_file']]);apis=[h['values'](self.env[s['env_file']]) for s in self.apis]
        require(self.worker_values.get('MODEL_NAME')==h['MODEL'] and self.worker_values.get('WORKER_MAX_INFLIGHT')=='64'
            and self.worker_values.get('WORKER_ADAPTIVE_KV_ENABLED','').lower() in {'true','1'},'adaptive235b64_required')

            and self.engine.get('gpu')=='0,1,2,3' and self.worker_values.get('INFERENCE_BASE_URLS')==f"http://127.0.0.1:{self.engine['port']}"
            and all(a.get('MODEL_NAME')==h['MODEL'] for a in apis)
            and all(apis[0].get(k,'60')==apis[1].get(k,'60') for k in ('SCHEDULER_AGING_SECONDS','SCHEDULER_WORKER_FRESHNESS_SECONDS')),'exact_model_engine_or_policy_knobs_required')
        self.records={};self.process_env={};self.argv={};bindings={}
        for s in self.all:
            key=self.key(s);record,env,argv,safe=self.reader(controller,s)
            require(all(env.get(k)==v for k,v in h['values'](self.env[s['env_file']]).items()),'effective_config_differs')
            self.records[key]=record;self.process_env[key]=env;self.argv[key]=argv;bindings[key]=safe

        self.drained=drained or (lambda:h['Deployment']._drained(self));self.idle=idle or self._idle

    @staticmethod
    def key(s):return s['role']+'-'+s['name']
    def save(self,value):self.h['save'](self.report_path,value);self.report=value
    def _idle(self):
        self.drained();self.h['Deployment']._profile(self)
        require(self.controller.engine_quiescence(self.engine).get('quiescent') is True,'engine_not_quiescent')
    def guard(self,stopping=None,warm=None):
        require(not self.unknown and not self.stop.is_set(),'unknown_or_interrupted_no_followon')
        if stopping:
            pending=self.report['pending_action'];require(stopping in {self.key(s) for s in [*self.apis,self.watchdog]}
                and pending and pending['stage']=='stop:'+stopping and pending['pid']==self.records[stopping]['pid']
                and pending['start_ticks']==self.records[stopping]['start_ticks'],'invalid_stopping_target')
        require(warm is None or warm in {self.key(s) for s in self.apis},'invalid_warming_target')
        require(self.h['private_file'](self.manifest)==self.manifest_raw and self.source_reader()==self.sources,'manifest_or_source_drift')
        require(all(self.h['private_file'](Path(p))==raw for p,raw in self.env.items()),'configuration_drift')
        for s in self.all:
            key=self.key(s)
            if key==stopping:continue  # role lock + pidfd + original PID-file checks own this one target
            require(self.controller.record(s)==self.records[key],'foreign_process_identity')
            if self.records[key] is None:continue
            record,env,argv,_=self.reader(self.controller,s)
            require(record==self.records[key] and env==self.process_env[key] and argv==self.argv[key],'effective_process_drift')
            if s in self.services and key!=warm:require(self.controller.responding(s,ready=True),'service_not_ready')
        for s in self.apis:
            require(self.h['private_file'](self.run/(s['name']+'.env.original'))==self.original[s['env_file']]
                and self.h['private_file'](self.run/(s['name']+'.env.staged'))==self.patched[s['env_file']],'private_snapshot_drift')
        self.drained()
    def action(self,stage,callback,**safe):
        self.guard();require(self.report['pending_action'] is None,'pending_no_replay')
        self.save({**self.report,'phase':'applying','pending_action':dict(stage=stage,at=self.h['stamp'](),**safe)})
        try:
            callback();self.save({**self.report,'pending_action':None,'events':[*self.report['events'],dict(stage=stage,confirmed_at=self.h['stamp'](),**safe)]})
        except BaseException:
            self.unknown=True;raise
    def stop_service(self,s):
        key=self.key(s);expected=self.records[key]
        def stop():
            check=lambda:self.guard(stopping=key)
            if s==self.watchdog:self.controller.graceful_stop(s,expected,30,check)
            else:self.api_stopper(self.root,self.controller,s,expected,check)
            require(self.controller.record(s) is None,'stop_not_confirmed');self.records[key]=None
        self.action('stop:'+key,stop,pid=expected['pid'],start_ticks=expected['start_ticks'])
    def start_service(self,s):
        key=self.key(s);require(self.records[key] is None and self.controller.record(s) is None,'start_requires_known_absence')
        def start():
            ack=self.starter(self.root,s,self.original_process_env[key]);record,env,argv,_=self.reader(self.controller,s)
            wanted=dict(self.original_process_env[key])
            if s in self.apis:wanted['SCHEDULER_POLICY']=self.target
            require(ack.get('status')=='started' and ack.get('pid')==record['pid'] and env==wanted and argv==self.argv[key],'unknown_start_ack_or_effective_policy')
            self.records[key]=record;self.process_env[key]=wanted
        self.action('start:'+key,start)
    def apply(self):
        require(self.report['phase']=='prepared' and not self.report['pending_action'],'existing_operation_no_replay')
        try:
            self.guard();self.idle();self.stop_service(self.watchdog)
            for s in self.apis:
                self.stop_service(s)
                def patch(s=s):
                    self.h['atomic_bytes'](Path(s['env_file']),self.patched[s['env_file']]);self.env[s['env_file']]=self.patched[s['env_file']]
                    require(self.h['private_file'](Path(s['env_file']))==self.env[s['env_file']],'policy_write_unconfirmed')
                self.action('policy:'+self.key(s),patch);self.start_service(s);deadline=self.clock()+45
                while True:
                    self.guard(warm=self.key(s))
                    if self.controller.responding(s,ready=True):break
                    require(self.clock()<deadline,'api_readiness_timeout');self.sleep(.5)
            self.guard();self.idle();self.start_service(self.watchdog);self.guard();self.idle()
            self.save({**self.report,'phase':'complete','finished_at':self.h['stamp'](),'passed':True,'watchdog_restored':True})
        except BaseException as error:
            try:self.save({**self.report,'phase':'manual_recovery_required','passed':False,'error_type':type(error).__name__,
                'automatic_rollback_or_replay':False,'watchdog_restart_after_failure':False})
            except BaseException:pass
            raise
        return self.report

def main(argv=None):

    p.add_argument('--operation',required=True);p.add_argument('--manifest',type=Path,default=ROOT/'.runtime/services-primary.json')

    try:

        require(bool(a.prepared_sha256)==a.apply,'apply_requires_exact_prepared_sha')
        require(re.fullmatch('[A-Za-z0-9][A-Za-z0-9_-]{0,63}',a.operation) is not None,'unique_operation_required')

        with ExitStack() as locks:

                locks.enter_context(locked(ROOT/'.runtime'/name))
            if not run.exists():require(not a.apply,'apply_requires_prepared');h['directory'](run,fresh=True);h['fsync_directory'](run.parent)
            else:h['directory'](run)
            if a.apply:require(sha(h['private_file'](run/'report.json',2*1024**2))==a.prepared_sha256,'exact_prepared_sha_required')
            stop=threading.Event()

            roll=Roll(ROOT,run,manifest,services,a.policy,h,WorkerController(root=ROOT),stop=stop)
            roll.guard();roll.idle()

            print(json.dumps(dict(phase=roll.report['phase'],report_sha256=sha(h['private_file'](run/'report.json',2*1024**2)),applied=a.apply)));return 0
    except BaseException as error:
        print(json.dumps(dict(refused_or_manual=True,error_code=str(error) if type(error) is Refusal else type(error).__name__,automatic_replay=False)));return 1
