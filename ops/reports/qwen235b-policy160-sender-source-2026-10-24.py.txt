def workload(U,rows,operation):
    selected={c:next((r for r in rows if r['custom_id']==name),None) for c,name in [('L','mix-0-2048-128'),('S','mix-0-128-32')]}

    for i,offset in enumerate(OFFSETS):
        c='L' if i<4 else 'S';source=selected[c];cap=128 if c=='L' else 32
        U.require(source.get('method')=='POST' and source.get('url')=='/v1/chat/completions' and source['body'].get('max_tokens')==cap,'fixture_cell_shape_differs')
        entries=[]
        for n in range(20):

        result.append(dict(index=i,class_label=c,offset=offset,rows=entries,raw=raw,sha256=sha(raw),bytes=len(raw)))

    U.require({str(r['version']):r['checksum'] for r in ledger}==context.ledger and len(ledger)==11,'exact_schema11_runtime_ledger_required')
    key=sha(context.bench_env['BATCHFORGE_API_KEY'].encode())

    U.require(limits and limits['value']['max_response_bytes']==65536 and limits['value']['max_error_bytes']==32768,'released_retention_caps_required')
    history=conn.execute("SELECT to_jsonb(p) AS value FROM tenant_history_retirement_policy p WHERE tenant_id='default'").fetchone()

def capacity(context,U,items,state):
    from psycopg.types.json import Jsonb
    conn=context.conn;bodies=[r['body'] for item in items for r in item['rows']]
    bodybytes=int(conn.execute('SELECT sum(public.batchforge_json_bytes(v)) AS n FROM jsonb_array_elements(%s) v',(Jsonb(bodies),)).fetchone()['n'])
    usage=conn.execute("""SELECT retained_tasks,charged_payload_bytes,(SELECT count(*) FROM tasks WHERE tenant_id='default') AS actual_tasks,
      (SELECT coalesce(sum(retention_body_bytes+retention_response_bytes+retention_error_bytes),0)::bigint FROM tasks WHERE tenant_id='default') AS actual_payload
      FROM tenant_retention_usage WHERE tenant_id='default'""").fetchone()
    U.require(usage and usage['retained_tasks']==usage['actual_tasks'] and usage['charged_payload_bytes']==usage['actual_payload'],'retention_counter_invariant_failed')
    need=bodybytes+160*(65536+4*32768);lim=state['limits']
    U.require(usage and usage['retained_tasks']+160<=lim['max_retained_tasks'] and usage['charged_payload_bytes']+need<=lim['max_payload_bytes'],'insufficient_task_or_payload_headroom')
    blob=conn.execute("SELECT stored_bytes,reserved_bytes,held_tasks FROM tenant_blob_usage WHERE tenant_id='default'").fetchone() or dict(stored_bytes=0,reserved_bytes=0,held_tasks=0)
    result=conn.execute("SELECT stored_bytes,reserved_bytes FROM tenant_result_usage WHERE tenant_id='default'").fetchone() or dict(stored_bytes=0,reserved_bytes=0)
    inputbytes=sum(i['bytes'] for i in items);api=context.api_env
    U.require(blob['stored_bytes']+blob['reserved_bytes']+inputbytes<=int(api.get('MAX_STORED_INPUT_BYTES_PER_TENANT',256*1024**2))
      and blob['held_tasks']+160<=int(api.get('MAX_QUEUED_TASKS_PER_TENANT',100000)),'insufficient_input_or_queue_headroom')
    U.require(result['stored_bytes']+result['reserved_bytes']+160*(65536+32768+4096)<=int(api.get('MAX_ARCHIVED_BYTES_PER_TENANT',1024**3)),'insufficient_archive_headroom')
    existing={r['resource']:r for r in conn.execute("SELECT resource,rows,bytes FROM tenant_auxiliary_usage WHERE tenant_id='default'").fetchall()}
    increments={'input_files':8,'upload_reservations':8,'idempotency':8,'blob_generations':8,'archive_runs':8,'archive_chunks':16,'input_read_leases':8,'result_read_leases':8}
    U.require(set(existing)==set(increments)=={r['resource'] for r in state['auxiliary']},'auxiliary_policy_resources_differ')
    for l in state['auxiliary']:
        u=existing[l['resource']];amount=increments[l['resource']];extra=amount*l['max_row_bytes'] if l['resource']=='idempotency' else 0
        U.require(u['rows']+amount<=l['max_rows'] and (not l['max_bytes'] or u['bytes']+extra<=l['max_bytes']),'insufficient_auxiliary_headroom')
        if l['resource']=='input_files':U.require(max(i['bytes'] for i in items)<=l['max_row_bytes'],'input_file_byte_ceiling')
    return dict(body_bytes=bodybytes,required_payload_bytes=need,input_bytes=inputbytes,retained_before=dict(usage))

def summarize(final,arrivals,terminal,expected):
    from datetime import datetime
    from scripts.benchmark import distribution
    batches={b['id']:b for b in final['batches']};tasks={t['id']:t for t in final['tasks']};result={}
    for kind in ('L','S'):
        members=[t for t in final['tasks'] if expected[t['custom_id']]==kind];attempts=[a for a in final['attempt_decisions'] if expected[tasks[a['task_id']]['custom_id']]==kind]
        indices=[str(i) for i in range(8) if ('L' if i<4 else 'S')==kind]
        queue=[(datetime.fromisoformat(a['claim_transaction_started_at'])-datetime.fromisoformat(batches[tasks[a['task_id']]['batch_id']]['created_at'])).total_seconds() for a in attempts]
        result[kind]=dict(tasks=len(members),prompt_tokens=sum(t['prompt_tokens'] for t in members),completion_tokens=sum(t['completion_tokens'] for t in members),
          DB_batch_created_to_task_finish_seconds=distribution([t['db_created_to_finish_seconds'] for t in members]),
          DB_batch_created_to_claim_transaction_start_seconds=distribution(queue),claim_transaction_start_to_SQL_finish_seconds=distribution([a['observed_attempt_seconds'] for a in attempts]),
          scheduled_to_client_admission_ACK_seconds=distribution([arrivals[i]['ack_offset']-OFFSETS[int(i)] for i in indices]),
          scheduled_to_client_batch_terminal_observation_seconds=distribution([terminal[i]['observed_offset']-OFFSETS[int(i)] for i in indices]))
    return dict(by_class=result,latency_semantics='DB start/update timestamps are not durable COMMIT; client terminal population is4 batches/class, SQL task population80/class')
