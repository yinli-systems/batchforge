from datetime import datetime, timezone
from pathlib import Path
import json, os, subprocess
import psycopg
from ops.supervise import load_manifest, private_read
from ops.ha.finalize_api_policy import values

def main():
    os.umask(0o077)
    root = Path('/root/batchforge-runtime/app')
    output = root / '.runtime/qwen235b-nccl-adaptive64-v1/topology-after-comparison.json'
    if output.exists():
        print(json.dumps({'existing':str(output)})); return
    _, services, _ = load_manifest(root / '.runtime/services-primary.json')
    env = values(private_read(Path(next(s for s in services if s['role']=='api')['env_file'])))
    with psycopg.connect(env['DATABASE_URL'],connect_timeout=5,
                        options='-c default_transaction_read_only=on -c statement_timeout=5000') as conn:
        active = conn.execute("SELECT count(*) FROM tasks WHERE status IN ('pending','running')").fetchone()[0]
    if active: raise RuntimeError('Wait for all platform cohorts to become terminal before topology observation')
    records = []
    for argv in (['nvidia-smi','topo','-m'], ['nvidia-smi','nvlink','--status']):
        try:
            result = subprocess.run(argv, capture_output=True, timeout=20, check=False)
            if len(result.stdout) + len(result.stderr) > 65536: raise RuntimeError('Topology output exceeds bound')
            records.append({'argv':argv,'returncode':result.returncode,
                'stdout':result.stdout.decode(errors='replace'),'stderr':result.stderr.decode(errors='replace')})
        except subprocess.TimeoutExpired:
            records.append({'argv':argv,'timed_out':True})
    report = {'observed_at':datetime.now(timezone.utc).isoformat(),'global_active_tasks':active,'observations':records,
              'limitations':['Read-only topology and available link state; not a bandwidth or latency benchmark',
                 'No inference requests or profiler activity generated by this observation']}
    with output.open('x') as stream: json.dump(report,stream,indent=2);stream.write('\n')
    print(json.dumps({'report':str(output),'observations':records}))

if __name__ == '__main__': main()
