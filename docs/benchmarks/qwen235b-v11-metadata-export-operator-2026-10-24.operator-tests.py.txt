 def child(self,argv,export_label='unit'):
  label=argv[argv.index('--round')+1];written='--write' in argv
  self.calls.append((label,written))
  current=self.pair/'metadata-export-operators'/export_label/'journal.json'
  journal=M['load'](current.read_bytes())
  self.assertEqual(journal['pending_action']['round'],label)
  if written:self.assertEqual(journal['pending_action']['stage'],'export_write')
  else:self.assertIn(journal['pending_action']['stage'],('prepare_validate','execute_validate'))
  receipt=M['load'](Path(self.f.manifest[label]['export_receipt']).read_bytes())
  if not written:receipt['metadata_files_written']={}
  if self.mode=='candidate_validate_fails' and label=='candidate' and not written and len(self.calls)>2:
   return dict(returncode=1,stdout=b'{"validated":false,"reason_code":"OperationalError"}\n',stderr_sha256=M['sha'](b''),stderr_bytes=0)
  if written:
   parent=self.pair/'exports'
   if not parent.exists():parent.mkdir(mode=0o700)
   target=parent/label;target.mkdir(mode=0o700)
   for index,(name,source) in enumerate(M['SOURCE_NAMES'].items()):
    path=target/(M['JOBS'][label]+'-'+name);path.write_bytes((self.runs/M['JOBS'][label]/source).read_bytes());path.chmod(0o600)
    if self.mode=='timeout_after_partial' and index==0:raise M['Refusal']('child_timeout_unknown_export_outcome')
    if self.mode=='success_with_partial_files':break
  raw=(json.dumps(receipt,indent=2)+'\n\n').encode();self.stdout[(label,written,len(self.calls))]=raw
  if self.mode=='invalid_stdout' and len(self.calls)>2:raw=b'SYNTHETIC_PRIVATE_BODY_NEVER_SAVE'
  return dict(returncode=0,stdout=raw,stderr_sha256=M['sha'](b'SYNTHETIC_DIAGNOSTIC'),stderr_bytes=20)
 def operator(self,label='unit'):return M['Operator'](label,runner=lambda argv:self.child(argv,label))
 def prepared(self):

 def test_success_preserves_exact_stdout_and_exact_fourteen_transport_files(self):

  for label in M['JOBS']:
   expectedstdout=next(v for (l,w,_),v in self.stdout.items() if l==label and w)
   self.assertEqual((root/'receipts'/(label+'-export-receipt.json')).read_bytes(),expectedstdout)

 def test_operator_lock_prevents_parallel_execute_and_journal_overwrite(self):
  expected=self.prepared();root=self.pair/'metadata-export-operators/unit';fd=open(root/'operation.lock','r+b');fcntl.flock(fd.fileno(),fcntl.LOCK_EX|fcntl.LOCK_NB);before=(root/'journal.json').read_bytes()

 def test_different_labels_cannot_execute_same_pair_concurrently(self):
  first=self.prepared();second=self.operator('alternate').prepare()['operator_prepared_sha256']
  alternate=self.pair/'metadata-export-operators/alternate';before=(alternate/'journal.json').read_bytes()
  entered=threading.Event();release=threading.Event();failures=[];results=[]
  def blocked_child(argv):
   entered.set()
   if not release.wait(5):raise RuntimeError('synthetic_test_wait_expired')
   return self.child(argv)
  def execute_first():
   try:results.append(M['Operator']('unit',runner=blocked_child).execute(first))
   except Exception as error:failures.append(error)
  worker=threading.Thread(target=execute_first);worker.start()
  try:
   self.assertTrue(entered.wait(5));calls=list(self.calls)
   with self.assertRaises(BlockingIOError):self.operator('alternate').execute(second)
   self.assertEqual(self.calls,calls);self.assertEqual((alternate/'journal.json').read_bytes(),before)
   self.assertFalse((alternate/'execution-marker.json').exists())
   self.assertFalse((self.pair/'exports').exists())
  finally:release.set();worker.join(10)
  self.assertFalse(worker.is_alive());self.assertFalse(failures);self.assertEqual(results[0]['phase'],'complete')
