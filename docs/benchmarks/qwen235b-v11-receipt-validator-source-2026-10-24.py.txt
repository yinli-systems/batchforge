def near(a,b,code='numeric_binding_differs'):

def local_path(value):

def read(path,maximum=16*1024**2):
 path=local_path(str(path));fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
 try:
  before=os.fstat(fd)
  require(stat.S_ISREG(before.st_mode) and before.st_uid==os.geteuid() and not before.st_mode&0o077
   and 0<before.st_size<=maximum,'unsafe_private_metadata')
  with os.fdopen(fd,'rb',closefd=False) as handle:raw=handle.read(maximum+1)
  after=os.fstat(fd)
  require(len(raw)==before.st_size and all(getattr(before,k)==getattr(after,k) for k in ('st_ino','st_dev','st_size','st_mtime_ns','st_ctime_ns')),'metadata_changed')
  return raw
 finally:os.close(fd)

def strict(raw):
 def unique(pairs):
  value={}
  for key,item in pairs:require(key not in value,'duplicate_metadata_key');value[key]=item
  return value
 def invalid(_):raise Refusal('nonfinite_metadata')
 return json.loads(raw.decode('utf-8'),object_pairs_hook=unique,parse_constant=invalid)

  'baseline':entry,'candidate':entry,'comparison':file,'pair_journal':file})
 require(manifest['version']==1 and manifest['operation']==OPERATION and manifest['prepared_plan_sha256']==PLAN_SHA,'exact_matched_operation_required')

   and pair[label+'_full_plan_passed'] is value['report']['passed'],'pair_round_report_or_passed_differs')

def write_private(path,raw):

 require(parent.resolve()==parent and stat.S_ISDIR(info.st_mode) and info.st_uid==os.geteuid() and not info.st_mode&0o077,'private_output_directory_required')
 fd=os.open(path,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
 with os.fdopen(fd,'wb') as handle:handle.write(raw);handle.flush();os.fsync(handle.fileno())
 fd=os.open(parent,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
 try:os.fsync(fd)
 finally:os.close(fd)
