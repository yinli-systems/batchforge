def require(value,code):
 if not value:raise ValueError(code)
def numeric(value):require(type(value) in (int,float) and math.isfinite(value),'nonfinite_numeric_metric');return value
def close(a,b):require(math.isclose(numeric(a),numeric(b),rel_tol=1e-10,abs_tol=1e-6),'recorded_prediction_arithmetic_differs')
def distribution(values):
 values=sorted(numeric(v) for v in values);require(bool(values),'empty_distribution')
 def q(p):
  n=(len(values)-1)*p;lo=math.floor(n);hi=math.ceil(n)
  return round(values[lo]+(values[hi]-values[lo])*(n-lo),6)
 return dict(samples=len(values),min=round(values[0],6),p50=q(.5),p95=q(.95),p99=q(.99),max=round(values[-1],6),mean=round(sum(values)/len(values),6))
def group(rows):return {'tasks':len(rows),'metrics':{k:distribution([r[k] for r in rows]) for k in rows[0]}}
