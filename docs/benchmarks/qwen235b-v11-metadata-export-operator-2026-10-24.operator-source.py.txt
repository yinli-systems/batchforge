def run_child(argv):
 """Bound both pipes; never save or print stderr plaintext."""
 process=subprocess.Popen(argv,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,cwd=ROOT)
 selector=selectors.DefaultSelector();buffers={'stdout':bytearray(),'stderr':bytearray()};deadline=time.monotonic()+120
 try:
  for name,pipe in [('stdout',process.stdout),('stderr',process.stderr)]:os.set_blocking(pipe.fileno(),False);selector.register(pipe,selectors.EVENT_READ,name)
  while selector.get_map():
   require(time.monotonic()<deadline,'child_timeout_unknown_export_outcome')
   for key,_ in selector.select(.2):
    chunk=os.read(key.fileobj.fileno(),8192)
    if not chunk:selector.unregister(key.fileobj);continue
    buffers[key.data].extend(chunk);require(len(buffers[key.data])<=65536,'child_output_bound_unknown_export_outcome')
  code=process.wait(timeout=5)
  return {'returncode':code,'stdout':bytes(buffers['stdout']),'stderr_sha256':sha(buffers['stderr']),'stderr_bytes':len(buffers['stderr'])}
 finally:
  selector.close()
  if process.poll() is None:
   process.terminate()
   try:process.wait(timeout=2)
   except subprocess.TimeoutExpired:process.kill();process.wait(timeout=2)
  for pipe in (process.stdout,process.stderr):pipe.close()

 def save(self):write(self.journal_path,encoded(self.journal),exclusive=False)
 def pending(self,stage,label=None):

 def confirmed(self,stage,checksum,label=None):

 def invoke(self,label,write_export,stage):

  argv=[sys.executable,str(ROOT/'.runtime/export_sustained_v11_metadata.py'),'--operation',OPERATION,'--round',label,'--prepared-sha256',PLAN_SHA]

  write(destination,result['stdout'])

  return value
 def fail(self,error):
  if self.journal is not None and self.armed:

   self.journal.update(phase='manual_recovery_required',failure_code=str(error) if isinstance(error,Refusal) else type(error).__name__,automatic_replay=False)
   try:self.save()
   except Exception:pass # Preserve the already-fsynced pending intent if storage failed.
 def prepare(self):

   and plan['operator_directory']==str(self.root) and plan['exports_root']==str(PAIR/'exports') and plan['interpreter']==str(Path(sys.executable).resolve()),'prepared_operator_configuration_differs')
  source_guard(plan['source_sha256'])
  require(self.journal['phase']=='prepared' and not self.journal.get('pending_action') and self.journal['operator_prepared_sha256']==expected_sha,'operator_not_clean_prepared')
  require(not (PAIR/'exports').exists() and not (PAIR/'exports').is_symlink(),'existing_exports_require_manual_review')
  self.armed=True;self.pending('execute_intent')

  self.journal['phase']='execute_validating';self.save()
  previews={label:self.invoke(label,False,'execute_validate') for label in JOBS}
  binding,pair_raw,comparison_raw=context(previews,self.loaded);require(binding==plan['binding'],'terminal_inputs_changed_since_prepare')

   require(not (PAIR/'exports'/label).exists() and not (PAIR/'exports'/label).is_symlink(),'existing_round_exports_require_manual_review')
   value=self.invoke(label,True,'export_write')
   require(value['source_inputs_sha256']==binding['artifact_sha256'][label],'written_export_differs_from_prepared_inputs')

   require({p.name for p in (PAIR/'exports'/label).iterdir()}=={JOBS[label]+'-'+n for n in SOURCE_NAMES},'export_directory_not_exact_five_files')
   require(all(sha(read(PAIR/'exports'/label/(JOBS[label]+'-'+n)))==h for n,h in binding['artifact_sha256'][label].items()),'written_export_actual_bytes_differs')
   self.confirmed('export_write',sha(read(self.root/'receipts'/(label+'-export-receipt.json'))),label)
  source_guard(plan['source_sha256']);current,_,_=context(previews,self.loaded);require(current==binding,'terminal_inputs_changed_during_export')

   'original_pair_journal_sha256':binding['pair_sha256'],'comparison_sha256':binding['comparison_sha256'],
   'files':files,'raw_outputs_or_env_or_dump_included':False,'business_state_modified':False,

  self.confirmed('transport_manifest_write',sha(body));self.journal.update(phase='complete',transport_manifest_sha256=sha(body),transport_file_count=14);self.save()
  return {'phase':'complete','transport_manifest_sha256':sha(body),'transport_manifest_path':str(self.root/'transport-manifest.json'),
